// Supporting mechanism screen only; never an end-to-end scoring substitute.
package connect

import (
	"context"
	"sync/atomic"
	"testing"
	"time"

	"github.com/urnetwork/server"
)

// Uses the actual constructor/default cache policy with one immutable lease.
// Every measured access must remain a hit; refresh/cold work invalidates this run.
func BenchmarkResidentContractWarmCacheParallel(b *testing.B) {
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	manager := newResidentContractManager(ctx, cancel, server.NewId(), DefaultExchangeSettings())
	source, destination := server.NewId(), server.NewId()
	var reads, refusals atomic.Int64
	manager.readContract = func(context.Context, server.Id, server.Id) residentContractAllowance {
		reads.Add(1)
		return residentContractAllowance{active: true, validUntil: time.Now().Add(time.Hour)}
	}
	if !manager.HasActiveContract(source, destination) {
		b.Fatal("initial positive read refused")
	}
	b.SetParallelism(16)
	b.ReportAllocs()
	b.ResetTimer()
	b.RunParallel(func(pb *testing.PB) {
		for pb.Next() {
			if !manager.HasActiveContract(source, destination) {
				refusals.Add(1)
			}
		}
	})
	b.StopTimer()
	if got := reads.Load(); got != 1 {
		b.Fatalf("not a pure warm-hit observation: source reads=%d", got)
	}
	if got := refusals.Load(); got != 0 {
		b.Fatalf("healthy cache hits refused=%d", got)
	}
	if err := manager.CloseAndWait(context.Background()); err != nil {
		b.Fatal(err)
	}
}
