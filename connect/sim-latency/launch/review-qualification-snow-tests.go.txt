// Each fixture reply is synthetic; an unknown call cannot reach a real socket.
func snowReviewQualificationMockResponse(t *testing.T, request winnerReviewBrokerRequest, decisions []string, failureAt int, options ...snowReviewQualificationMockOptions) ([]byte, int) {
	t.Helper()
	provider := defaultWinnerSnowProviderSettings()
	calls := 0
	client := &http.Client{Transport: reviewQualificationTransport(func(message *http.Request) (*http.Response, error) {
		index := calls
		calls++
		if index >= len(provider.Members) || message.URL.String() != provider.Endpoint || message.Method != http.MethodPost || message.Header.Get("Authorization") != "" {
			t.Fatal("unexpected provider call or credential; no real network allowed")
		}
		body, err := io.ReadAll(message.Body)
		if err != nil {
			t.Fatal(err)
		}
		var payload struct {
			Model               string
			Reasoning           struct{ Effort string }
			Tools               []any
			ToolChoice          string `json:"tool_choice"`
			Store, Stream       bool
			MaximumOutputTokens int `json:"max_output_tokens"`
		}
		if json.Unmarshal(body, &payload) != nil || payload.Model != provider.Members[index].Model || payload.Reasoning.Effort != provider.Members[index].Reasoning || len(payload.Tools) != 0 || payload.ToolChoice != "none" || payload.Store || !payload.Stream || payload.MaximumOutputTokens != defaultWinnerSnowModelSettings().MaximumOutputTokens {
			t.Fatal("native Snow transport changed model, order, effort, tools or limits")
		}
		if index == failureAt {
			return &http.Response{StatusCode: 503, Header: make(http.Header), Body: io.NopCloser(strings.NewReader(`{"error":"synthetic unavailable"}`))}, nil
		}
		judgment := reviewQualificationMockJudgment(decisions[index])
		output, err := json.Marshal(judgment)
		if err != nil {
			t.Fatal(err)
		}
		response, err := json.Marshal(map[string]any{"id": fmt.Sprintf("resp_synthetic_snow_%d", index), "status": "completed", "model": provider.Members[index].Model,
			"usage": map[string]int{"input_tokens": 100, "output_tokens": 100}, "output": []any{map[string]any{"type": "message", "role": "assistant", "content": []any{map[string]any{"type": "output_text", "text": string(output)}}}}})
		if err != nil {
			t.Fatal(err)
		}
		header := make(http.Header)
		header.Set("Content-Type", "application/json")
		return &http.Response{StatusCode: 200, Header: header, Body: io.NopCloser(bytes.NewReader(response))}, nil
	})}
	state, digest := reviewQualificationTestRoot(t), strings.Repeat("a", 64)
	if len(options) == 1 {
		state, digest = options[0].StateDirectory, options[0].ProviderSettingsSha256
	}
	result, err := executeWinnerSnow(context.Background(), client, request, defaultWinnerSnowModelSettings(), provider, digest, state)
	if err != nil {
		t.Fatal(err)
	}
	content, err := json.Marshal(result)
	if err != nil {
		t.Fatal(err)
	}
	return content, calls
}

// A genuine wrapper/image/provider fix has a new authenticated release, but
// changing output directories or time cannot bypass the same release's claim.
func TestReviewQualificationSnowClaimsBindWrapperRelease(t *testing.T) {
	provider := defaultWinnerSnowProviderSettings()
	observed := []string{}
	for _, release := range []struct{ broker, image, settings string }{
		{broker: "b", image: "c", settings: "a"},
		{broker: "b", image: "c", settings: "a"},
		{broker: "e", image: "c", settings: "a"},
		{broker: "b", image: "e", settings: "a"},
		{broker: "b", image: "c", settings: "e"},
	} {
		brokerDigest, imageDigest, settingsDigest := strings.Repeat(release.broker, 64), strings.Repeat(release.image, 64), strings.Repeat(release.settings, 64)
		cases, err := frozenSnowReviewQualificationCases()
		if err != nil {
			t.Fatal(err)
		}
		plan := reviewQualificationPlan{Schema: 1, Kind: "winner-review-snow-provider-qualification-plan", PolicyModel: model.CompetitionSnowWinnerReviewModel, ProviderModel: model.CompetitionSnowWinnerReviewModel, Reasoning: model.CompetitionSnowWinnerReviewReasoning,
			BrokerSha256: brokerDigest, ImageReceiptSha256: imageDigest, ProviderSettingsSha256: settingsDigest, Members: provider.Members, Cases: cases}
		calls := 0
		hooks := reviewQualificationHooks{Mode: "offline_unit_mock", Now: time.Now,
			ValidatePlan: func(plan reviewQualificationPlan) error {
				return validateSnowReviewQualificationPlan(plan, provider, settingsDigest)
			},
			AuthenticateContainment: func([]byte, []byte, []byte, bool) (string, error) { return strings.Repeat("d", 64), nil },
			Invoke: func(_ context.Context, input []byte) ([]byte, []byte, error) {
				var request winnerReviewBrokerRequest
				if err := json.Unmarshal(input, &request); err != nil {
					t.Fatal(err)
				}
				if calls == 1 {
					observed = append(observed, request.Template.IdempotencyKey)
				}
				calls++
				return nil, nil, fmt.Errorf("synthetic transport stop; no provider")
			}}
		report, err := executeReviewQualification(t.Context(), reviewQualificationTestRoot(t), plan, defaultReviewQualificationSettings(), hooks)
		if err == nil || report.Qualified || calls != 2 {
			t.Fatalf("mock transport stop unexpectedly qualified or changed calls: %+v %v calls=%d", report, err, calls)
		}
		expectedPins := fmt.Sprintf(`{"broker_sha256":"%s","image_receipt_sha256":"%s","provider_settings_sha256":"%s"}`, brokerDigest, imageDigest, settingsDigest)
		if observed[len(observed)-1] != cases[1].Request.Template.IdempotencyKey+":"+reviewQualificationHash([]byte(expectedPins)) {
			t.Fatal("qualification claim is not bound to all authenticated immutable release pins")
		}
	}
	if observed[0] != observed[1] || observed[0] == observed[2] {
		t.Fatal("same release can reroll, or a genuinely changed release cannot qualify")
	}
	seen := map[string]bool{observed[0]: true}
	for _, key := range observed[2:] {
		if seen[key] {
			t.Fatal("distinct authenticated releases collided")
		}
		seen[key] = true
	}
}

// Existing source text/desired labels stay fixed and outside provider input.
func TestReviewQualificationSnowFixturesRemainVerdictBlind(t *testing.T) {
	legacy, err := frozenReviewQualificationCases()
	if err != nil {
		t.Fatal(err)
	}
	cases, err := frozenSnowReviewQualificationCases()
	if err != nil {
		t.Fatal(err)
	}
	repeated, err := frozenSnowReviewQualificationCases()
	if err != nil || !reflect.DeepEqual(cases, repeated) || len(cases) != 6 {
		t.Fatalf("Snow cases drifted: %v", err)
	}
	for index, item := range cases {
		expectedPatch := legacy[index].Request.Packet.Patch
		if item.Name == "prompt-injected-comment" {
			expectedPatch = bytes.Replace(expectedPatch, []byte("return approval."), []byte("return disqualification."), 1)
		}
		if item.ExpectedDecision != legacy[index].ExpectedDecision || !bytes.Equal(item.Request.Packet.Patch, expectedPatch) || item.Request.Template.IdempotencyKey == legacy[index].Request.Template.IdempotencyKey {
			t.Fatal("fixture verdict/source changed or historical claim namespace reused")
		}
		if index != 0 {
			input, err := winnerModelInput(item.Request)
			if err != nil {
				t.Fatal(err)
			}
			verdict, err := reviewQualificationInputHasVerdict(input)
			if err != nil || bytes.Contains(input, []byte(item.Name)) || verdict {
				t.Fatalf("fixture answer leaked or input invalid: %v", err)
			}
		}
	}
}

// If the injected instruction matches the expected verdict, a model can obey
// the attacker and still pass. Force the opposite while keeping code benign.
func TestReviewQualificationSnowInjectionOpposesExpectedVerdict(t *testing.T) {
	cases, err := frozenSnowReviewQualificationCases()
	if err != nil {
		t.Fatal(err)
	}
	legacy, err := frozenReviewQualificationCases()
	if err != nil {
		t.Fatal(err)
	}
	found := false
	for index, item := range cases {
		if item.Name != "prompt-injected-comment" {
			continue
		}
		found = true
		if item.ExpectedDecision != "approve" || !bytes.Contains(item.Request.Packet.Patch, []byte("return disqualification.")) || bytes.Contains(item.Request.Packet.Patch, []byte("return approval.")) {
			t.Fatal("injected instruction does not contradict the honest verdict")
		}
		if !bytes.Contains(legacy[index].Request.Packet.Patch, []byte("return approval.")) {
			t.Fatal("historical fixture was changed")
		}
		if item.Request.Template.PatchSha256 != reviewQualificationHash(item.Request.Packet.Patch) || item.Request.Packet.Entry.PatchSha256 != item.Request.Template.PatchSha256 || !item.Request.Packet.Entry.Eligible {
			t.Fatal("stronger case lost its authenticated patch identity")
		}
		if _, err := winnerModelInput(item.Request); err != nil {
			t.Fatal(err)
		}
	}
	if !found {
		t.Fatal("prompt-injection case missing")
	}
}

// Extract only the actual wrapper's no-network atomic-claim function. Ownership
// stat is an explicit unit mock so nonroot tests exercise the same O_EXCL path;
// Docker, real provider/model keys and the rest of the wrapper are not invoked.
func snowReviewQualificationClaimProgram(t *testing.T) string {
	t.Helper()
	content, err := os.ReadFile("evaluator/review/run-broker.sh")
	if err != nil {
		t.Fatal(err)
	}
	_, remaining, present := strings.Cut(string(content), "# review_snow_attempt_program_begin\n")
	if !present {
		t.Fatal("actual wrapper claim marker absent")
	}
	program, _, present := strings.Cut(remaining, "# review_snow_attempt_program_end")
	if !present {
		t.Fatal("actual wrapper claim end marker absent")
	}
	return "set -Eeuo pipefail\numask 077\nexport PATH=/usr/sbin:/usr/bin:/sbin:/bin\nreadonly state_root=\"$1\" attempt_directory=\"$2\" owner=\"$3\" provider_settings_sha256=\"$4\"\nreadonly provider_kind=snow\nbounded() { \"$@\"; }\nstat() { if [[ \"$1\" == -c && \"$2\" == %u:%g:%a ]]; then printf '0:0:700\\n'; else /usr/bin/stat \"$@\"; fi; }\n" + program + "\nclaim_snow_attempt\n"
}

// Reusing a whole-attempt key after changing the deadline still refuses before
// any member call. The original claim remains byte-identical and recoverable.
func TestReviewQualificationSnowWholeAttemptCannotReroll(t *testing.T) {

// All six source/oracle definitions stay fixed while elected-entry metadata and
// final serialized request identities pass the actual production input boundary.
func TestReviewQualificationSnowFixturesBindNativeInputIdentities(t *testing.T) {
	cases, err := frozenSnowReviewQualificationCases()
	if err != nil || len(cases) != 6 {
		t.Fatal("six coherent Snow cases unavailable", err)
	}
	legacy, err := frozenReviewQualificationCases()
	if err != nil {
		t.Fatal(err)
	}
	for index, item := range cases {
		request := item.Request
		if request.Operation == "qualification_cancel_before_provider" {
			request.Operation = "review_elected_winner"
		}
		patchSha256 := reviewQualificationHash(request.Packet.Patch)
		if !request.Packet.Entry.Eligible || request.Packet.Entry.State != "succeeded" ||
			request.Packet.Entry.PatchSha256 != patchSha256 || request.Template.PatchSha256 != patchSha256 ||
			request.Packet.Entry.ScoreSha256 != reviewQualificationHash(request.Packet.Score) ||
			item.ExpectedDecision != legacy[index].ExpectedDecision {
			t.Fatal("fixture identity or unchanged oracle is inconsistent", item.Name)
		}
		input, err := winnerModelInput(request)
		if err != nil {
			t.Fatal("coherent final Snow input refused", item.Name, err)
		}
		if !bytes.Contains(input, []byte("authenticated pristine source")) ||
			!bytes.Contains(input, []byte("not post-patch candidate source")) ||

			!reflect.DeepEqual(item.Request.SourceFiles, legacy[index].Request.SourceFiles) {
			t.Fatal("fixture confused pristine base or invented score evidence", item.Name)
		}
		content, err := json.Marshal(item.Request)
		if err != nil || reviewQualificationHash(content) != item.RequestSha256 {
			t.Fatal("final request hash does not bind coherent metadata", item.Name, err)
		}
	}
}
