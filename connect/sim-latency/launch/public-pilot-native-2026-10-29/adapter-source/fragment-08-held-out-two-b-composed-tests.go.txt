// Counts only explicit deterministic fixture reads; no crypto/rand source is
// reachable through this fixture or its fake process hooks.
type heldOutTwoBFixtureEntropy struct {
	Reader io.Reader
	Calls  int
}

// The reader is instance-owned and used serially by the preparation core.
func (self *heldOutTwoBFixtureEntropy) Read(content []byte) (int, error) {
	self.Calls++
	return self.Reader.Read(content)
}

// Captured root entries are checked by the unchanged full-ancestry validator
// before and after an independently bounded/digest-checked fixture read.
func (self *heldOutTwoBCompositionFixture) readRootPin(pin heldOutPinnedFile, mode os.FileMode, maximum int64) ([]byte, error) {
	self.RuntimeReads++
	stat := func(path string) (heldOutRootPathMetadata, error) {
		metadata, ok := self.RootMetadataKVs[path]
		if !ok {
			return heldOutRootPathMetadata{}, errors.New("synthetic root entry absent")
		}
		return metadata, nil
	}
	if err := validateHeldOutRootPath(pin.Path, mode, stat); err != nil {
		return nil, err
	}
	content, ok := self.RuntimeContentBytesKVs[pin.Path]
	if !ok || maximum <= 0 || int64(len(content)) > maximum || heldOutDigest(content) != pin.Sha256 {
		return nil, errors.New("synthetic root bytes changed or exceed their bound")
	}
	if err := validateHeldOutRootPath(pin.Path, mode, stat); err != nil {
		return nil, err
	}
	return bytes.Clone(content), nil
}

// Files also exist in the fixture tree for ordinary script-pin validation.
// Captured metadata is deliberately separate from real t.TempDir ancestry.
func (self *heldOutTwoBCompositionFixture) captureRootFile(t *testing.T, path string, content []byte, mode os.FileMode) heldOutPinnedFile {
	t.Helper()
	pin := heldOutOpWrite(t, path, content)
	self.RuntimeContentBytesKVs[path] = bytes.Clone(content)
	self.RootMetadataKVs[path] = heldOutRootPathMetadata{Mode: mode, Uid: 0, Gid: 0, Links: 1}
	for parent := filepath.Dir(path); ; parent = filepath.Dir(parent) {
		self.RootMetadataKVs[parent] = heldOutRootPathMetadata{Mode: os.ModeDir | 0555, Uid: 0, Gid: 0, Links: 2}
		if parent == string(filepath.Separator) {
			break
		}
	}
	return pin
}

// Invalid input must fail in the same composed entry before a private attempt,

		t.Fatalf("%s reached preparation or invocation: plan=%v reads=%d generated=%d invoked=%d err=%v", label, plan != nil, self.Entropy.Calls, self.Preparation.Generated, self.Invocations, err)

		}
	}

		name   string
		mutate func(*heldOutTwoBCompositionFixture)
	}{
		{name: "component digest", mutate: func(f *heldOutTwoBCompositionFixture) {
			path := filepath.Join(f.Preparation.Release.ScriptDirectory, "evaluator.sh")
			f.RuntimeContentBytesKVs[path] = append(bytes.Clone(f.RuntimeContentBytesKVs[path]), '\n')
		}},
		{name: "root leaf owner", mutate: func(f *heldOutTwoBCompositionFixture) {
			path := f.Preparation.Release.Runtime.Manifest.Path
			metadata := f.RootMetadataKVs[path]
			metadata.Uid = 1000
			f.RootMetadataKVs[path] = metadata
		}},
		{name: "root leaf link count", mutate: func(f *heldOutTwoBCompositionFixture) {
			path := f.Preparation.Release.Runtime.RunnerManifest.Path

			metadata.Links = 2
			f.RootMetadataKVs[path] = metadata
		}},
		{name: "writable ancestor", mutate: func(f *heldOutTwoBCompositionFixture) {
			path := filepath.Dir(f.Preparation.Release.Runtime.Manifest.Path)
			metadata := f.RootMetadataKVs[path]
			metadata.Mode = os.ModeDir | 0775
			f.RootMetadataKVs[path] = metadata

		{name: "symlink leaf", mutate: func(f *heldOutTwoBCompositionFixture) {

	}
}

// The second immutable-input fence follows generation and private-plan write.
// A late replacement keeps that exact failed attempt without public output/reroll.
func TestHeldOutTwoBComposedRetainsLateReplacementWithoutReroll(t *testing.T) {
	cases := []struct {
		name string
		pin  func(*heldOutTwoBCompositionFixture) heldOutPinnedFile
	}{
		{name: "approval", pin: func(f *heldOutTwoBCompositionFixture) heldOutPinnedFile {
			return f.Preparation.Config.TwoBAddendum.Approval
		}},
		{name: "reference", pin: func(f *heldOutTwoBCompositionFixture) heldOutPinnedFile { return f.References[1].Direction }},
		{name: "bridge", pin: func(f *heldOutTwoBCompositionFixture) heldOutPinnedFile {
			return f.Preparation.Config.Amendment.BridgeReceipt
		}},
	}
	for _, testCase := range cases {
		fixture := newHeldOutTwoBCompositionFixture(t)
		op := fixture.Preparation
		inputs, err := fixture.authenticate()
		if err != nil {
			t.Fatal(testCase.name, err)
		}
		pin := testCase.pin(fixture)
		original, err := os.ReadFile(pin.Path)
		if err != nil {
			t.Fatal(testCase.name, err)
		}
		checks := 0
		public, err := prepareHeldOutCampaign(t.Context(), op.Config, heldOutDigest(op.ConfigBytes), *inputs, op.Release, op.Settings, op.Hooks, func() error {
			checks++
			if checks == 2 {
				heldOutOpWrite(t, pin.Path, append(bytes.Clone(original), '\n'))
			}
			again, err := fixture.authenticate()
			if err != nil {
				return err
			}
			if !reflect.DeepEqual(again, inputs) {
				return errors.New("late replacement changed authenticated inputs")
			}
			return nil
		})
		if err == nil || public != nil || checks != 2 || fixture.Entropy.Calls == 0 || op.Generated != 2 || fixture.Invocations != 0 {
			t.Fatalf("%s late substitution did not fail the final fence: checks=%d generated=%d invoked=%d err=%v", testCase.name, checks, op.Generated, fixture.Invocations, err)
		}
		for _, name := range []string{"preparation.started", "configuration.json", "private-plan.json"} {
			if _, err := os.Lstat(filepath.Join(op.Config.PrivateRoot, name)); err != nil {
				t.Fatal(testCase.name, "lost its exact failed private attempt", name, err)
			}
		}
		if _, err := os.Lstat(op.Config.PublicPlanPath); !errors.Is(err, os.ErrNotExist) {
			t.Fatal(testCase.name, "published a changed-input campaign", err)
		}
		draws := fixture.Entropy.Calls
		heldOutOpWrite(t, pin.Path, original)
		if _, err := fixture.authenticate(); err != nil {
			t.Fatal(testCase.name, "did not restore the original base", err)
		}
		if _, err := prepareHeldOutCampaign(t.Context(), op.Config, heldOutDigest(op.ConfigBytes), *inputs, op.Release, op.Settings, op.Hooks, func() error {
			_, err := fixture.authenticate()
			return err
		}); err == nil || fixture.Entropy.Calls != draws || op.Generated != 2 || fixture.Invocations != 0 {
			t.Fatal(testCase.name, "restored approval silently permitted a replacement draw", err)
		}
	}
}
