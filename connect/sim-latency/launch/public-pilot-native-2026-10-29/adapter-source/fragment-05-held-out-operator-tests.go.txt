// Owns only generated test paths and identities; no retained production data.
type heldOutOpFixture struct {
	Root                       string
	Config                     heldOutPreparationConfig
	ConfigBytes                []byte
	Inputs                     heldOutPreparationInputs
	Release                    heldOutRelease
	Settings                   heldOutOperatorSettings
	Hooks                      heldOutPreparationHooks
	Generated                  int
	Reauthenticated            int
	ConstructorAuthentications int
	PrivatePin, PublicPin      heldOutPinnedFile
}

// Test writing is intentionally replaceable so mutations exercise pinned-byte
// refusal; actual operator publication uses exclusive, fsynced writes instead.
func heldOutOpWrite(t *testing.T, path string, content []byte) heldOutPinnedFile {
	t.Helper()
	if err := os.MkdirAll(filepath.Dir(path), 0700); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(path, content, 0600); err != nil {
		t.Fatal(err)
	}
	return heldOutPinnedFile{Path: path, Sha256: heldOutDigest(content)}
}

// Encodes explicit zero/false fields too, matching the strict field census.
func heldOutOpJson(t *testing.T, path string, value any) heldOutPinnedFile {
	t.Helper()
	content, err := json.MarshalIndent(value, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	return heldOutOpWrite(t, path, content)
}

// Synthetic ids are deterministic and visibly generated, never campaign ids.
func heldOutOpId(value int) string {
	return fmt.Sprintf("%08x-0000-4000-8000-%012x", value, value)
}

// Only fresh fixture receipt construction shares this immutable actual-ELF
// identity. Production authentication never reads this cache and still streams
// the actual executable for every qualification check.
var heldOutFreshFixtureExecutableIdentity struct {
	Once   sync.Once
	Sha256 string
	Err    error
}

// Reads the current Go test ELF, never a fabricated executable or helper.
func readHeldOutFixtureExecutableDigest() (string, error) {
	executable, err := os.Executable()
	if err != nil {
		return "", err
	}
	file, err := os.Open(executable)
	if err != nil {
		return "", err
	}
	digest := sha256.New()
	if _, err := io.Copy(digest, file); err != nil {
		_ = file.Close()
		return "", err
	}
	if err := file.Close(); err != nil {
		return "", err
	}
	return hex.EncodeToString(digest.Sum(nil)), nil
}

// Reusing a receipt identity saves only fixture generation work, never a native
// verifier call. Ordinary old fixture constructors retain uncached behavior.
func heldOutFixtureExecutableDigest(t *testing.T, cacheForFresh bool) string {
	t.Helper()
	if cacheForFresh {
		heldOutFreshFixtureExecutableIdentity.Once.Do(func() {
			heldOutFreshFixtureExecutableIdentity.Sha256, heldOutFreshFixtureExecutableIdentity.Err = readHeldOutFixtureExecutableDigest()
		})
		if heldOutFreshFixtureExecutableIdentity.Err != nil {
			t.Fatal(heldOutFreshFixtureExecutableIdentity.Err)
		}
		return heldOutFreshFixtureExecutableIdentity.Sha256
	}
	digest, err := readHeldOutFixtureExecutableDigest()
	if err != nil {
		t.Fatal(err)
	}
	return digest
}

// Keeps the independently supplied configuration pin in sync for semantic
// mutations. Old pins are retained separately by tests of byte substitution.
func (self *heldOutOpFixture) refreshConfig(t *testing.T) {
	t.Helper()
	content, err := json.MarshalIndent(self.Config, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	self.ConfigBytes = content
	self.Hooks.Configuration = content
}

// The synthetic callback still reauthenticates every report/source/reference
// file on both sides of preparation, independently of the native workload hook.
func (self *heldOutOpFixture) prepare(t *testing.T) *heldOutPrivatePlan {
	t.Helper()
	public, err := prepareHeldOutCampaign(t.Context(), self.Config, heldOutDigest(self.ConfigBytes), self.Inputs,
		self.Release, self.Settings, self.Hooks, func() error {
			self.Reauthenticated++
			_, err := authenticateHeldOutPreparation(self.Config, self.Release, self.Settings)
			return err
		})
	if err != nil {
		t.Fatal(err)
	}
	self.PrivatePin = heldOutPinnedFile{Path: filepath.Join(self.Config.PrivateRoot, "private-plan.json"), Sha256: public.PrivatePlanSha256}
	publicBytes, err := os.ReadFile(self.Config.PublicPlanPath)
	if err != nil {
		t.Fatal(err)
	}
	self.PublicPin = heldOutPinnedFile{Path: self.Config.PublicPlanPath, Sha256: heldOutDigest(publicBytes)}
	privateBytes, err := readHeldOutPinned(self.PrivatePin, self.Settings.MaximumBytes, true)
	if err != nil {
		t.Fatal(err)
	}
	var plan heldOutPrivatePlan
	if err := json.Unmarshal(privateBytes, &plan); err != nil {
		t.Fatal(err)
	}
	return &plan
}

// The full composed path generates five actual native workloads and fifteen
// requests before publishing any public commitment. Nothing is evaluated.
func TestHeldOutOperatorPreparesCompleteNativeFiveRoundPlan(t *testing.T) {
	fixture := newHeldOutOpFixture(t)
	plan := fixture.prepare(t)
	if fixture.Generated != 5 || fixture.Reauthenticated != 2 || len(plan.Rounds) != 5 {
		t.Fatal("incomplete generation or before/after authentication")
	}
	if _, err := verifyHeldOutPlan(fixture.PrivatePin, fixture.PublicPin, fixture.Release, fixture.Settings); err != nil {
		t.Fatal(err)
	}
	for _, round := range plan.Rounds {
		if len(round.Public.Jobs) != 3 {
			t.Fatal("partial reference order")
		}
		for _, job := range round.Public.Jobs {
			root := heldOutJobRoot(fixture.Config.PrivateRoot, round.Public, job)
			for _, name := range []string{"precommitted-request.json", "providers.yml", "canonical.patch"} {
				info, err := os.Stat(filepath.Join(root, "artifacts", name))
				if err != nil || info.Mode().Perm() != 0600 {
					t.Fatal("private publication mode changed")
				}
			}
			if _, err := os.Lstat(filepath.Join(root, "started.json")); !errors.Is(err, os.ErrNotExist) {
				t.Fatal("preparation started evaluation")
			}
		}
	}
}

// Whitespace drift still violates the independently published accepted-report
// byte pin, including drift after all private artifacts were already written.
func TestHeldOutOperatorRefusesChangedReportAtFinalPublication(t *testing.T) {
	fixture := newHeldOutOpFixture(t)
	calls := 0
	_, err := prepareHeldOutCampaign(t.Context(), fixture.Config, heldOutDigest(fixture.ConfigBytes), fixture.Inputs, fixture.Release, fixture.Settings, fixture.Hooks, func() error {
		calls++
		if calls == 2 {
			content, readErr := os.ReadFile(fixture.Config.StageAReport.Path)
			if readErr != nil {
				return readErr
			}
			if err := os.WriteFile(fixture.Config.StageAReport.Path, append(content, '\n'), 0600); err != nil {
				return err
			}
		}
		_, err := authenticateHeldOutMargin(fixture.Config, fixture.Release, fixture.Settings)
		return err
	})
	if err == nil || !strings.Contains(err.Error(), "final publication") || calls != 2 {
		t.Fatalf("changed accepted report not refused at final publication: %v", err)
	}
	if _, err := os.Lstat(fixture.Config.PublicPlanPath); !errors.Is(err, os.ErrNotExist) {
		t.Fatal("partial public plan escaped")
	}
	if _, err := os.Stat(filepath.Join(fixture.Config.PrivateRoot, "private-plan.json")); err != nil {
		t.Fatal("failed preparation evidence was discarded")
	}
}

// Entropy failure leaves an exclusive started record. Retrying the same root
// cannot quietly draw a new campaign after a partial preparation.
func TestHeldOutOperatorEntropyFailureCannotRerollRoot(t *testing.T) {
	fixture := newHeldOutOpFixture(t)
	fixture.Hooks.Entropy = bytes.NewReader(nil)
	for i := 0; i < 2; i++ {
		if _, err := prepareHeldOutCampaign(t.Context(), fixture.Config, heldOutDigest(fixture.ConfigBytes), fixture.Inputs, fixture.Release, fixture.Settings, fixture.Hooks, func() error { return nil }); err == nil {
			t.Fatal("failed entropy or reused private root accepted")
		}
	}
	if fixture.Generated != 0 {
		t.Fatal("native generator ran without campaign entropy")
	}
	if _, err := os.Stat(filepath.Join(fixture.Config.PrivateRoot, "preparation.started")); err != nil {
		t.Fatal("failed draw evidence removed")
	}
}

// Injects a repeated effective seed directly, without searching for a real hash
// collision; the composed census must refuse before any public plan exists.
func TestHeldOutOperatorRejectsRepeatedEffectiveWorkload(t *testing.T) {
	fixture := newHeldOutOpFixture(t)
	generate := fixture.Hooks.Generate
	var firstSeed int64
	fixture.Hooks.Generate = func(ctx context.Context, seed string, policy model.CompetitionEvaluationPolicy, settings heldOutWorkloadSettings) (*heldOutGeneratedWorkload, error) {
		workload, err := generate(ctx, seed, policy, settings)
		if err != nil {
			return nil, err
		}
		if firstSeed == 0 {
			firstSeed = workload.EffectiveSeed
		} else {
			workload.EffectiveSeed = firstSeed
		}
		return workload, nil
	}
	_, err := prepareHeldOutCampaign(t.Context(), fixture.Config, heldOutDigest(fixture.ConfigBytes), fixture.Inputs, fixture.Release, fixture.Settings, fixture.Hooks, func() error { return nil })
	if err == nil || !strings.Contains(err.Error(), "repeated or excluded effective seed") {
		t.Fatalf("repeated effective workload accepted: %v", err)
	}
}

// Test fixtures reach the pin boundary with valid private roots. Symlink/FIFO
// refusal is synchronous metadata validation, not a timeout-based assertion.
func TestHeldOutOperatorRefusesLinkedSpecialAndPublicPrivateInputs(t *testing.T) {
	root := newCalibrationFileRoot(t)
	pin := heldOutOpWrite(t, filepath.Join(root, "input.json"), []byte("synthetic private bytes\n"))
	alias := filepath.Join(root, "alias.json")
	if err := os.Symlink(pin.Path, alias); err != nil {
		t.Fatal(err)
	}
	if _, err := readHeldOutPinned(heldOutPinnedFile{Path: alias, Sha256: pin.Sha256}, 1024, true); err == nil {
		t.Fatal("symlink alias accepted")
	}
	fifo := filepath.Join(root, "fifo")
	if err := syscall.Mkfifo(fifo, 0600); err != nil {
		t.Fatal(err)
	}
	if _, err := readHeldOutPinned(heldOutPinnedFile{Path: fifo, Sha256: pin.Sha256}, 1024, true); err == nil {
		t.Fatal("fifo accepted")
	}
	if err := os.Chmod(pin.Path, 0644); err != nil {
		t.Fatal(err)
	}
	if _, err := readHeldOutPinned(pin, 1024, true); err == nil {
		t.Fatal("public mode on private seed/request file accepted")
	}
}

// A completeness flag cannot hide an omitted prior raw/effective/provider or
// commitment identity when the exact retained source still contains it.
func TestHeldOutOperatorAuthenticatesEveryExclusionSourceIdentity(t *testing.T) {
	fixture := newHeldOutOpFixture(t)
	content, err := os.ReadFile(fixture.Config.Exclusions.Path)
	if err != nil {
		t.Fatal(err)
	}
	var original heldOutExclusions
	if err := json.Unmarshal(content, &original); err != nil {
		t.Fatal(err)
	}
	if err := validateHeldOutExclusions(original, fixture.Settings.MaximumBytes); err != nil {
		t.Fatal(err)
	}
	for _, mutate := range []func(*heldOutExclusions){
		func(e *heldOutExclusions) { e.PrivateSeedHexes = []string{fmt.Sprintf("%064x", 9002)} },
		func(e *heldOutExclusions) { e.ProviderDigests = []string{strings.Repeat("f", 64)} },
		func(e *heldOutExclusions) { e.RetiredCommitments = []string{strings.Repeat("f", 64)} },
		func(e *heldOutExclusions) { e.EffectiveSeeds = append(append([]int64{}, e.EffectiveSeeds...), 9002) },
	} {
		changed := original
		mutate(&changed)
		if err := validateHeldOutExclusions(changed, fixture.Settings.MaximumBytes); err == nil || !strings.Contains(err.Error(), "census") {
			t.Fatalf("unsupported or omitted exclusion identity accepted: %v", err)
		}
	}
}

// Legacy effective seeds exceed the exact integer range of float64. Parsing a
// source through interface{} must not round one seed into a neighboring value.
func TestHeldOutOperatorExclusionCensusPreservesInt64Precision(t *testing.T) {
	fixture := newHeldOutOpFixture(t)
	content, err := os.ReadFile(fixture.Config.Exclusions.Path)
	if err != nil {
		t.Fatal(err)
	}
	var exclusions heldOutExclusions
	if err := json.Unmarshal(content, &exclusions); err != nil {
		t.Fatal(err)
	}
	const legacySeed int64 = 9007199254740993
	legacy := heldOutOpJson(t, filepath.Join(fixture.Root, "inputs", "large-legacy-seed.json"), map[string]any{"generator_seed": legacySeed})
	exclusions.Sources = append(exclusions.Sources, legacy)
	exclusions.EffectiveSeeds = append(exclusions.EffectiveSeeds, legacySeed)
	if err := validateHeldOutExclusions(exclusions, fixture.Settings.MaximumBytes); err != nil {
		t.Fatal(err)
	}
	exclusions.EffectiveSeeds[len(exclusions.EffectiveSeeds)-1] = legacySeed + 1
	if err := validateHeldOutExclusions(exclusions, fixture.Settings.MaximumBytes); err == nil {
		t.Fatal("neighboring rounded legacy seed accepted")
	}
}
