// Exercises the actual asynchronous shell reader without Docker or providers.
// An implicit background stdin becomes /dev/null even with a finite parent pipe.
func TestReviewQualificationWrapperPreservesBackgroundInput(t *testing.T) {
	content, err := os.ReadFile("evaluator/review/run-broker.sh")
	if err != nil {
		t.Fatal(err)
	}
	_, program, found := strings.Cut(string(content), "# review_input_capture_program_begin\n")
	if !found {
		t.Fatal("wrapper input capture boundary missing")
	}
	program, _, found = strings.Cut(program, "# review_input_capture_program_end")
	if !found {
		t.Fatal("wrapper input capture end missing")
	}
	for _, input := range []string{`{"operation":"synthetic_cancel"}`, strings.Repeat("x", 64), strings.Repeat("y", 65), strings.Repeat("z", 66)} {
		directory := t.TempDir()
		command := exec.CommandContext(t.Context(), "/bin/bash", "-c", "set -Eeuo pipefail\ninput_limit=64\nattempt_directory=\"$1\"\n"+program, "synthetic-input-capture", directory)
		command.Stdin = strings.NewReader(input)
		output, commandErr := command.CombinedOutput()
		captured, readErr := os.ReadFile(filepath.Join(directory, "input.json"))
		if readErr != nil || string(captured) != input[:min(len(input), 65)] {
			t.Fatalf("background reader lost request bytes: got %q want %q; read=%v command=%v output=%s", captured, input, readErr, commandErr, output)
		}
		if (commandErr == nil) != (len(input) <= 64) {
			t.Fatalf("input limit changed: bytes=%d error=%v output=%s", len(input), commandErr, output)
		}
	}
}

// Implements only the HTTP test seam; it has no real provider credential.
func (self reviewQualificationTransport) RoundTrip(request *http.Request) (*http.Response, error) {
	return self(request)
}

// Generates all six complete findings with explicit synthetic mock provenance.
func reviewQualificationMockJudgment(expected string) winnerModelJudgment {
	judgment := winnerModelJudgment{Decision: expected, ReasonCodes: []string{"APPROVED"}, PublicExplanation: "Synthetic unit judgment; tests were not executed.", PrivateEvidence: "Offline deterministic model mock; not a real provider.", Findings: []model.CompetitionReviewFinding{}}
	for _, criterion := range []string{"measurement_integrity", "workload_generalization", "containment", "patch_scope", "reproducibility", "source_safety"} {
		judgment.Findings = append(judgment.Findings, model.CompetitionReviewFinding{Criterion: criterion, Disposition: "pass", ReasonCode: "APPROVED", Explanation: "Synthetic complete criterion context."})
	}
	if expected == "disqualify" {
		judgment.ReasonCodes = []string{"MEASUREMENT_FABRICATION"}
		judgment.Findings[0].Disposition, judgment.Findings[0].ReasonCode = "fail", "MEASUREMENT_FABRICATION"
	}
	if expected == "uncertain" {
		judgment.ReasonCodes = []string{"SOURCE_NOT_READY"}
		for i := range judgment.Findings {
			judgment.Findings[i].Disposition, judgment.Findings[i].ReasonCode = "uncertain", "SOURCE_NOT_READY"
		}
	}
	return judgment
}

// Runs the actual native provider/parser/result path over fixed SSE bytes, not
// an invented winnerReviewBrokerResult with skipped identity/authority checks.
func reviewQualificationMockResponse(t *testing.T, request winnerReviewBrokerRequest, expected string) []byte {
	t.Helper()
	judgment := reviewQualificationMockJudgment(expected)
	output, err := json.Marshal(judgment)
	if err != nil {
		t.Fatal(err)
	}
	event, err := json.Marshal(map[string]any{"type": "response.completed", "response": map[string]any{"id": "resp_synthetic_qualification", "status": "completed", "model": winnerReviewMiniProviderSnapshot, "usage": map[string]int{"input_tokens": 100, "output_tokens": 100}, "output": []any{map[string]any{"type": "message", "role": "assistant", "content": []any{map[string]any{"type": "output_text", "text": string(output)}}}}}})
	if err != nil {
		t.Fatal(err)
	}
	client := &http.Client{Transport: reviewQualificationTransport(func(message *http.Request) (*http.Response, error) {
		if message.URL.String() != "https://api.openai.com/v1/responses" || message.Method != http.MethodPost || message.Header.Get("Authorization") != "Bearer synthetic-qualification-key" {
			t.Fatal("provider authority changed")
		}
		body, err := io.ReadAll(message.Body)
		if err != nil {
			t.Fatal(err)
		}
		var payload struct {
			Model         string
			Reasoning     struct{ Effort string }
			Tools         []any
			ToolChoice    string `json:"tool_choice"`
			Store, Stream bool
		}
		if json.Unmarshal(body, &payload) != nil || payload.Model != winnerReviewMiniProviderSnapshot || payload.Reasoning.Effort != "xhigh" || len(payload.Tools) != 0 || payload.ToolChoice != "none" || payload.Store || !payload.Stream || bytes.Contains(body, []byte("synthetic-qualification-key")) {
			t.Fatal("qualification provider request gained tools/authority or lost exact model/effort")
		}
		return &http.Response{StatusCode: 200, Header: make(http.Header), Body: io.NopCloser(bytes.NewReader(append(append([]byte("data: "), event...), []byte("\n\ndata: [DONE]\n\n")...)))}, nil
	})}
	result, err := executeWinnerModel(context.Background(), client, "synthetic-qualification-key", request, defaultWinnerModelSettings())
	if err != nil {
		t.Fatal(err)
	}
	content, err := json.Marshal(result)
	if err != nil {
		t.Fatal(err)
	}
	return content
}

// Complete offline success must remain explicitly non-provider qualification.
func TestReviewQualificationOfflineSuccessCannotQualifyProvider(t *testing.T) {
	plan := newReviewQualificationTestPlan(t)
	calls := 0
	clock := time.Now().UTC().Add(24 * time.Hour)
	hooks := reviewQualificationHooks{Mode: "offline_unit_mock", Now: func() time.Time { return clock }, AuthenticateContainment: func([]byte, []byte, []byte, bool) (string, error) { return strings.Repeat("d", 64), nil }, Invoke: func(_ context.Context, input []byte) ([]byte, []byte, error) {
		calls++
		var request winnerReviewBrokerRequest
		if json.Unmarshal(input, &request) != nil {
			t.Fatal("bad dispatched request")
		}
		if calls == 1 {
			return nil, []byte("synthetic canceled wrapper"), errors.New("synthetic cancellation before provider input")
		}
		return reviewQualificationMockResponse(t, request, plan.Cases[calls-1].ExpectedDecision), []byte("synthetic containment only"), nil
	}}

		t.Fatalf("offline mock became live qualification or lost cases: %+v %v", report, err)
	}
	for _, observation := range report.Observations {
		if observation.Name == "insufficient-transitive-context" && observation.WouldAuthorize {
			t.Fatal("uncertain mock became authority")
		}
	}
}

// Native HTTP refusal parsing preserves quota metadata/body privately and never
// turns a nonzero exit into a winner review or calls the remaining cases.
func TestReviewQualificationQuotaRefusalStopsWithoutRetry(t *testing.T) {
	plan := newReviewQualificationTestPlan(t)
	calls := 0
	clock := time.Now().UTC().Add(24 * time.Hour)
	root := reviewQualificationTestRoot(t)
	hooks := reviewQualificationHooks{Mode: "offline_unit_mock", Now: func() time.Time { return clock }, AuthenticateContainment: func([]byte, []byte, []byte, bool) (string, error) { return strings.Repeat("d", 64), nil }, Invoke: func(_ context.Context, input []byte) ([]byte, []byte, error) {
		calls++
		if calls == 1 {
			return nil, nil, errors.New("synthetic cancellation")
		}
		var request winnerReviewBrokerRequest
		if json.Unmarshal(input, &request) != nil {
			t.Fatal("bad request")
		}
		client := &http.Client{Transport: reviewQualificationTransport(func(*http.Request) (*http.Response, error) {
			return &http.Response{StatusCode: 429, Header: http.Header{"X-Request-Id": []string{"req_synthetic_quota"}}, Body: io.NopCloser(strings.NewReader(`{"error":{"code":"credit_balance_exhausted","type":"insufficient_quota","message":"synthetic unavailable credit"}}`))}, nil
		})}
		_, err := executeWinnerModel(t.Context(), client, "synthetic-key", request, defaultWinnerModelSettings())
		var refusal *winnerReviewProviderRefusal
		if !errors.As(err, &refusal) {
			t.Fatal("typed refusal lost")
		}
		content, marshalErr := json.Marshal(struct {
			Schema  int                          `json:"schema"`
			Kind    string                       `json:"kind"`
			Refusal *winnerReviewProviderRefusal `json:"provider_refusal"`
			Body    json.RawMessage              `json:"private_redacted_body"`
		}{Schema: 1, Kind: "review_provider_refusal", Refusal: refusal, Body: refusal.PrivateRedactedBody})
		if marshalErr != nil {
			t.Fatal(marshalErr)
		}
		return content, nil, err
	}}
	report, err := executeReviewQualification(t.Context(), root, plan, defaultReviewQualificationSettings(), hooks)
	if err == nil || calls != 2 || report.Qualified || report.Complete || len(report.Observations) != 2 || report.Observations[1].WouldAuthorize || report.Observations[1].Refusal == nil || report.Observations[1].Refusal.Code != "credit_balance_exhausted" {
		t.Fatalf("quota was retried, reclassified or approved: %+v %v", report, err)
	}
	if _, err := os.Stat(filepath.Join(root, "case-02", "broker.stdout")); err != nil {
		t.Fatal("private quota envelope not retained")
	}
	if _, err := executeReviewQualification(t.Context(), root, plan, defaultReviewQualificationSettings(), hooks); err == nil || calls != 2 {
		t.Fatal("same failed attempt was rerun")
	}
}

// Exact field identity distinguishes an operator's expected verdict from the
// legitimate digest of the already-published reward decision. Strings remain
// candidate data rather than being reinterpreted as JSON member names.
func reviewQualificationInputHasVerdict(content []byte) (bool, error) {
	var value any
	if err := json.Unmarshal(content, &value); err != nil {
		return false, err
	}
	var contains func(any) bool
	contains = func(value any) bool {
		switch value := value.(type) {
		case map[string]any:
			for key, item := range value {
				if key == "expected_decision" || contains(item) {
					return true
				}
			}
		case []any:
			for _, item := range value {
				if contains(item) {
					return true
				}
			}
		}
		return false
	}
	return contains(value), nil
}

// A decision digest is authority context, not the answer a reviewer should
// produce; exact verdict fields at any nesting depth remain forbidden.
func TestReviewQualificationSeparatesVerdictFromDecisionDigest(t *testing.T) {
	for _, fixture := range []struct {
		content string
		verdict bool
	}{
		{content: `{"context":{"template":{"expected_decision_sha256":"synthetic-digest"}}}`},
		{content: `{"patch_text":"candidate comment mentions expected_decision"}`},
		{content: `{"expected_decision":"approve"}`, verdict: true},
		{content: `{"context":[{"expected_decision":"disqualify"}]}`, verdict: true},
	} {
		verdict, err := reviewQualificationInputHasVerdict([]byte(fixture.content))
		if err != nil || verdict != fixture.verdict {
			t.Errorf("verdict member detection for %s: %v, %v", fixture.content, verdict, err)
		}
	}
	if _, err := reviewQualificationInputHasVerdict([]byte(`{"expected_decision":`)); err == nil {
		t.Fatal("malformed JSON treated as verdict-blind evidence")
	}
}

// Real kernel rule text, including quoted comments and canonical /32 provider
// addresses, must have one permitted destination followed by an unconditional drop.
func TestReviewQualificationFenceRequiresProviderOnlyDefaultDeny(t *testing.T) {
	owner := strings.Repeat("a", 64)
	chain := "URRV_" + owner[:16]
	subnet, provider := "192.168.240.0/24", "192.0.2.10"
	comment := "urnetwork-review-" + owner
	forward := []byte(fmt.Sprintf("-N DOCKER-USER\n-A DOCKER-USER -s %s -m comment --comment \"%s\" -j %s\n", subnet, comment, chain))
	input := []byte(fmt.Sprintf("-P INPUT ACCEPT\n-A INPUT -s %s -m comment --comment \"%s\" -j DROP\n", subnet, comment))
	rules := []byte(fmt.Sprintf("-N %s\n-A %s -d %s/32 -p tcp -m tcp --dport 443 -j ACCEPT\n-A %s -j DROP\n", chain, chain, provider, chain))
	if err := validateReviewQualificationFence(rules, forward, input, owner, subnet, provider); err != nil {
		t.Fatal(err)
	}
	if validateReviewQualificationFence(rules, forward, input, "", subnet, provider) == nil {
		t.Fatal("missing fence owner accepted")
	}
	for _, changed := range [][]byte{bytes.Replace(rules, []byte("--dport 443"), []byte("--dport 80"), 1), bytes.Replace(rules, []byte("-j DROP"), []byte("-j ACCEPT"), 1), append(append([]byte{}, rules...), []byte("-A "+chain+" -j ACCEPT\n")...)} {
		if validateReviewQualificationFence(changed, forward, input, owner, subnet, provider) == nil {
			t.Fatal("widened actual provider firewall accepted")
		}
	}
}

// Executes only the actual wrapper's extracted awk parser on synthetic text,
// never the wrapper, Docker, a key file, namespace or provider request.
func reviewQualificationShellMappedUid(t *testing.T, content []byte) (uint64, error) {
	t.Helper()
	settings := defaultReviewQualificationUidSettings()
	if len(content) > settings.MaximumBytes {
		return 0, errors.New("synthetic UID capture exceeds wrapper bound")
	}
	wrapper, err := os.ReadFile(filepath.Join("evaluator", "review", "run-broker.sh"))
	if err != nil {
		t.Fatal(err)
	}
	prefix := "readonly review_uid_map_program='\n"
	suffix := "\n'\n# review_uid_map_program_end"
	start := strings.Index(string(wrapper), prefix)
	if start < 0 {
		t.Fatal("qualified wrapper UID parser source missing")
	}
	start += len(prefix)
	end := strings.Index(string(wrapper)[start:], suffix)
	if end < 0 {
		t.Fatal("qualified wrapper UID parser terminator missing")
	}
	program := string(wrapper)[start : start+end]
	path := filepath.Join(reviewQualificationTestRoot(t), "synthetic-uid-map.txt")
	if err := os.WriteFile(path, content, 0600); err != nil {
		t.Fatal(err)
	}
	ctx, cancel := context.WithTimeout(t.Context(), defaultReviewQualificationSettings().CleanupGrace)
	defer cancel()
	command := exec.CommandContext(ctx, "awk", program, path)
	output, err := command.Output()
	if err != nil {
		return 0, err
	}
	value, err := strconv.ParseUint(strings.TrimSpace(string(output)), 10, 32)
	if err != nil || string(output) != fmt.Sprintf("%d\n", value) {
		return 0, errors.New("wrapper UID parser output is not one exact unsigned decimal")
	}
	return value, nil
}

// No hardcoded host offset can substitute for the actual running child map.
func TestReviewQualificationUidMapAcceptsActualMappedIdentity(t *testing.T) {
	for _, item := range []struct {
		body     string
		expected uint64
	}{{body: "         0          0 4294967295\n", expected: 65532}, {body: "0 2400000 65536\n", expected: 2465532}, {body: "0 8000000 1024\n65000 6200000 1000\n", expected: 6200532}, {body: "65532 7300000 1\n", expected: 7300000}} {
		mapped, err := reviewQualificationMappedUid([]byte(item.body), 65532)
		if err != nil || mapped != item.expected {
			t.Fatalf("Go UID map mismatch: got=%d expected=%d err=%v", mapped, item.expected, err)
		}
		shellMapped, err := reviewQualificationShellMappedUid(t, []byte(item.body))
		if err != nil || shellMapped != item.expected {
			t.Fatalf("actual wrapper UID parser mismatch: got=%d expected=%d err=%v", shellMapped, item.expected, err)
		}
	}
}

// Kernel-format bounds reject malformed/negative/overflow ranges before either
// decoder can convert an invalid number into a plausible key owner.
func TestReviewQualificationUidMapRejectsMalformedOrOverflow(t *testing.T) {
	tooMany := strings.Builder{}
	tooMany.WriteString("65532 2400000 1\n")
	for index := 0; index < defaultReviewQualificationUidSettings().MaximumRanges; index++ {
		fmt.Fprintf(&tooMany, "%d %d 1\n", index, 8000000+index)
	}
	for _, body := range []string{"", "\n", "0 2400000\n", "0 2400000 65536 extra\n", "-1 2400000 65536\n", "+0 2400000 65536\n", "00 2400000 65536\n", "0 2400000 0\n", "0 2400000 65536\r\n", "0 4294967290 65536\n", "4294967295 2400000 1\n", "0 2400000 4294967296\n", tooMany.String(), strings.Repeat(" ", defaultReviewQualificationUidSettings().MaximumBytes) + "0 2400000 65536\n"} {

		}
	}
