// Production reducers, rather than an accepted boolean, construct the expected
// complete healthy slower result. Every id and scalar is synthetic.
func newHeldOutPublicPilotCohort() (calibrationAnalysisCohort, calibrationAnalysisSettings) {
	settings := defaultCalibrationAnalysisSettings()
	settings.Cohorts, settings.MarginFloor = 1, .269
	cohort := newCalibrationAnalysisFixture(settings)[0]
	for index := range cohort.Candidates {
		cohort.Candidates[index].diagnostics.RawScore = 130 + float64(index)
	}
	rebuildCalibrationAnalysisFixture(&cohort)
	return cohort, settings
}

// A non-significant adverse direction is sufficient only when every native
// gate/reducer passes; significance and the selected practical margin stay intact.
func TestHeldOutPublicPilotAcceptsActualMeanAndMedianSlower(t *testing.T) {
	cohort, settings := newHeldOutPublicPilotCohort()
	before, err := json.Marshal(cohort.Result.Significance)
	if err != nil {
		t.Fatal(err)
	}
	if err := validateHeldOutPublicPilotDirection(cohort, settings); err != nil {
		t.Fatal(err)
	}
	after, err := json.Marshal(cohort.Result.Significance)
	if err != nil || !bytes.Equal(before, after) || cohort.Result.Significance.StatisticallySignificant {
		t.Fatal("direction check changed significance or required significant worsening", err)
	}
}

// Mean and median are independent predeclared conditions; neither alone wins.
func TestHeldOutPublicPilotRejectsMeanOnlyWorsening(t *testing.T) {
	cohort, settings := newHeldOutPublicPilotCohort()
	for index := range cohort.Candidates {
		value := 80.0
		if index >= 6 {
			value = 200
		}
		cohort.Candidates[index].diagnostics.RawScore = value
	}
	rebuildCalibrationAnalysisFixture(&cohort)
	if cohort.Result.Significance.CandidateMeanRawScore <= cohort.Result.Significance.BaselineMeanRawScore ||
		validateHeldOutPublicPilotDirection(cohort, settings) == nil {
		t.Fatal("mean-only worsening accepted or synthetic root cause absent")
	}
}

// A worse median with an improved run-level mean remains an unfavorable pilot.
func TestHeldOutPublicPilotRejectsMedianOnlyWorsening(t *testing.T) {
	cohort, settings := newHeldOutPublicPilotCohort()

	}
}

// The deliberately minimal YAML is census-only synthetic data, not a runnable
// pilot provider profile or a substitute for the preflight generator validator.
func newHeldOutPublicPilotMergeFixture(t *testing.T) heldOutPublicPilotMergeFixture {
	t.Helper()
	original := newHeldOutFreshExclusionBuilderFixture(t)
	beforePin, err := prepareHeldOutFreshSingleExclusions(original.Config, original.Release, original.Settings)
	if err != nil {
		t.Fatal(err)
	}
	content, err := readHeldOutPinned(beforePin, original.Settings.MaximumBytes, false)
	if err != nil {
		t.Fatal(err)
	}
	var before heldOutExclusions
	if err := decodeHeldOutComplete(content, &before, "synthetic before catalog"); err != nil {
		t.Fatal(err)
	}
	providers := heldOutOpWrite(t, filepath.Join(original.Root, "synthetic-public-providers.yml"),
		[]byte(fmt.Sprintf("seed: %d\n", original.Facts.PublicPilotSeed)))
	request := heldOutRequest{RoundSeedHex: fmt.Sprintf("%064x", original.Facts.PublicPilotSeed),
		ProvidersPath: providers.Path, ProvidersSha256: providers.Sha256}
	requestPin := heldOutOpJson(t, filepath.Join(original.Root, "synthetic-public-request.json"), request)
	plan := heldOutPublicPilotPlan{Schema: 1, Kind: "sim-latency-fresh-public-regression-pilot-plan", Protocol: heldOutProtocolFreshSingleB,
		PublicSeed: original.Facts.PublicPilotSeed, ProvidersSha256: providers.Sha256, Request: requestPin}
	planPin := heldOutOpJson(t, filepath.Join(original.Root, "synthetic-public-plan.json"), plan)
	return heldOutPublicPilotMergeFixture{Before: before, Request: request, Plan: plan, PlanPin: planPin, Providers: providers,
		Reviewed: append(append([]heldOutPinnedFile{}, before.Sources...), requestPin, providers, planPin),
		At:       original.Config.ReviewedAt.Add(time.Hour), Original: original}
}

// All historical sets survive, the direct seed/raw label/provider digest are
// added exactly, and no invented public seed commitment enters the catalog.
func TestHeldOutPublicPilotAfterMergePreservesCompleteNativeCensus(t *testing.T) {
	self := newHeldOutPublicPilotMergeFixture(t)
	merged, err := mergeHeldOutPublicPilotExclusions(self.Before, self.Request, self.Plan, self.PlanPin, self.Providers,
		self.Reviewed, self.At, self.Original.Settings.MaximumBytes)
	if err != nil || len(merged.PrivateSeedHexes) != len(self.Before.PrivateSeedHexes)+1 ||
		len(merged.EffectiveSeeds) != len(self.Before.EffectiveSeeds)+1 || len(merged.ProviderDigests) != len(self.Before.ProviderDigests)+1 ||
		!reflect.DeepEqual(merged.RetiredCommitments, self.Before.RetiredCommitments) {
		t.Fatal("after-public merge omitted identities or invented a commitment", err)
	}
	if err := validateHeldOutExclusions(*merged, self.Original.Settings.MaximumBytes); err != nil {
		t.Fatal(err)
	}
}

// Captured Git/memory process boundaries leave every native plan/provider/
// policy/build/source validator active over wholly synthetic owned artifacts.
type heldOutPublicPilotAdmissionFixture struct {
	Base                     *heldOutFreshSingleFixture
	Plan                     heldOutPublicPilotPlan
	Config                   heldOutPublicPilotExecutionConfig
	PublishedContentBytesKVs map[string][]byte
	Clock                    time.Time
	Invocations              int
}

// Each artifact is independently bound to an existing publication schema; the
// fake reader supplies only the committed bytes, never a permissive boolean.
func (self *heldOutPublicPilotAdmissionFixture) publish(t *testing.T, artifact heldOutPinnedFile, at time.Time) heldOutPinnedFile {
	t.Helper()
	content, err := readHeldOutPinned(artifact, self.Base.Base.Preparation.Settings.MaximumBytes, false)
	if err != nil {
		t.Fatal(err)
	}
	relative := fmt.Sprintf("synthetic-published-%02d.json", len(self.PublishedContentBytesKVs))
	self.PublishedContentBytesKVs[relative] = content
	op := self.Base.Base.Preparation
	return heldOutOpJson(t, filepath.Join(op.Root, relative+".publication.json"), heldOutPublication{
		Schema: 1, Kind: "sim-latency-held-out-plan-publication", PlanSha256: artifact.Sha256, PublishedAt: at,
		Commit: strings.Repeat("c", 40), Repository: op.Root, PlanPath: relative, RemoteRef: "refs/remotes/origin/synthetic",
		PulledPushed: true, Evidence: []heldOutPinnedFile{self.Plan.Tool}})
}

// No process is executed. Unexpected calls fail rather than silently simulating
// an evaluator, Docker metadata, scoring or any native source-context operation.
func (self *heldOutPublicPilotAdmissionFixture) read(_ context.Context, program string, arguments ...string) ([]byte, error) {
	if program == self.Base.Facts.MemoryExecutable.Path {
		return []byte(`{"mode":"memory-preflight-only","required_management_bytes":8589934592,"evaluation_limit_bytes":103079215104,"evidence_limit_bytes":5368709120,"management_reserve_bytes":25769803776}`), nil
	}
	if program != "/usr/bin/git" {
		return nil, errors.New("unexpected synthetic prerequisite program")
	}
	for index, argument := range arguments {
		if argument == "merge-base" {
			return []byte{}, nil
		}
		if argument == "show" && index+1 < len(arguments) {
			_, path, ok := strings.Cut(arguments[index+1], ":")
			if ok {
				if content, found := self.PublishedContentBytesKVs[path]; found {
					return bytes.Clone(content), nil
				}
			}
		}
	}
	return nil, errors.New("unexpected synthetic Git prerequisite")
}

// Alternate fixed public integers change only wholly synthetic provider input;
// the native integer mapping and original two-position ancestry stay separate.
func newHeldOutPublicPilotAdmissionFixtureForSeed(t *testing.T, publicSeed int64) *heldOutPublicPilotAdmissionFixture {
	t.Helper()
	base := newHeldOutFreshSingleFixture(t)
	op := base.Base.Preparation
	oldPublicSeed := base.Facts.PublicPilotSeed
	base.Facts.PublicPilotSeed = publicSeed
	for index, seed := range base.Exclusions.EffectiveSeeds {
		if seed == oldPublicSeed {
			base.Exclusions.EffectiveSeeds[index] = publicSeed
		}
	}
	// The older composed fixture intentionally needed only public ancestry.
	// This adapter also reconstructs the exact strict native builder output.
	oldPrivate := base.Facts.OriginalPrivatePlan
	privateBytes, err := readHeldOutPinned(oldPrivate, op.Settings.MaximumBytes, true)
	if err != nil {
		t.Fatal(err)
	}
	var original heldOutPrivatePlan
	if err := json.Unmarshal(privateBytes, &original); err != nil {
		t.Fatal(err)
	}
	original.Config.Protocol, original.Config.PrivateRoot = heldOutProtocolThreeABridgeTwoB, base.Facts.OriginalRoot
	original.Config.GeneratorFiles, original.Config.Scripts = []heldOutGeneratorFile{}, []heldOutPinnedFile{}
	original.References = []heldOutReference{}
	base.Facts.OriginalPrivatePlan = heldOutOpJson(t, oldPrivate.Path, original)
	publicBytes, err := readHeldOutPinned(base.Facts.OriginalPublicPlan, op.Settings.MaximumBytes, false)
	if err != nil {
		t.Fatal(err)
	}
	var public heldOutPublicPlan
	if err := decodeHeldOutComplete(publicBytes, &public, "synthetic original public ancestry"); err != nil {

// Returns the exact frozen policy flags used by the synthetic raw runs.
func heldOutPublicPilotRawFlags(policy model.CompetitionEvaluationPolicy) map[string]string {
	flags := map[string]string{"impair": strconv.FormatBool(policy.ImpairmentEnabled), "fleet_shards": strconv.Itoa(policy.FleetShards),
		"hosts": strconv.Itoa(policy.ExchangeHosts), "site_listen": policy.SiteListen, "api_port": strconv.Itoa(policy.ApiPort), "reset": "true"}
	for name, milliseconds := range map[string]int64{"ramp": policy.RampMs, "prewarm": policy.PrewarmMs, "settle": policy.SettleMs,

	}
	return flags
}

// The captured process boundary recomputes from copies of the authenticated raw
// files. Only path-bearing fields/marker digests change in that private copy;
// original reports stay byte-for-byte pinned and are rechecked by the job reader.
func (self *heldOutPublicPilotRawR9Fixture) replay(t *testing.T) calibrationScoreReplay {
	t.Helper()
	return func(ctx context.Context, root string, inputs ScoreInputs, image string, roundId string, baseline bool) ([]byte, error) {
		a := self.Admission
		op := a.Base.Base.Preparation
		want := self.Candidate
		if baseline {
			want = self.Control
		}
		if root != self.Request.ArtifactDirectory || image != op.Release.ImageId || roundId != self.Request.RoundId || !reflect.DeepEqual(inputs, want) {
			return nil, errors.New("captured native replay changed its actual root/input/image/round binding")
		}
		self.ReplayCalls++
		originalCorpus := filepath.Join(root, "evidence", "scorer-input")
		copiedCorpus := filepath.Join(newCalibrationFileRoot(t), "captured-namespace")
		if err := copyCalibrationReplayCorpus(ctx, originalCorpus, copiedCorpus, defaultCalibrationAnalysisSettings()); err != nil {
			return nil, err
		}
		mapHostPath := func(path string) string {
			relative, err := filepath.Rel(originalCorpus, path)
			if err != nil || relative == ".." || strings.HasPrefix(relative, ".."+string(filepath.Separator)) {
				t.Fatal("captured replay input escaped its authenticated corpus", err)
			}
			return filepath.Join(copiedCorpus, relative)
		}
		mapped := ScoreInputs{BaselineManifest: mapHostPath(inputs.BaselineManifest)}
		for _, field := range []struct {
			from []string
			to   *[]string
		}{{from: inputs.Runs, to: &mapped.Runs}, {from: inputs.Stderr, to: &mapped.Stderr},
			{from: inputs.Accounting, to: &mapped.Accounting}, {from: inputs.Samples, to: &mapped.Samples},
			{from: inputs.ResourceReports, to: &mapped.ResourceReports}, {from: inputs.Markers, to: &mapped.Markers}} {
			for _, path := range field.from {
				*field.to = append(*field.to, mapHostPath(path))
			}
		}
		for _, run := range mapped.Runs {
			stats, _, err := readOfficialRunSidecar(run)
			if err != nil {
				return nil, err
			}
			for _, path := range []*string{&stats.StatsRoot, &stats.AccountingReport, &stats.ResourceReport, &stats.FinalMarker} {
				if !strings.HasPrefix(*path, "/artifacts/") || filepath.Clean(*path) != *path {
					return nil, errors.New("captured native sidecar did not retain its original namespace")
				}
				*path = filepath.Join(copiedCorpus, strings.TrimPrefix(*path, "/artifacts/"))
			}
			if err := writeRunStats(filepath.Join(filepath.Dir(run), "run.json"), stats); err != nil {
				return nil, err
			}
			if err := writeFinalMarker(stats.FinalMarker, filepath.Join(filepath.Dir(run), "run.json"), stats); err != nil {
				return nil, err
			}
		}
		var output []byte
		var err error
		if baseline {
			value, buildErr := BuildScoreBaseline(ScoreBaselineInputs{Runs: mapped.Runs, Stderr: mapped.Stderr, Accounting: mapped.Accounting,
				Samples: mapped.Samples, ResourceReports: mapped.ResourceReports, Markers: mapped.Markers,
				RoundId: roundId, TakeoverMargin: self.Request.EvaluationPolicy.TakeoverMargin})
			if buildErr != nil {
				return nil, buildErr
			}
			output, err = marshalScoreBaseline(value)
		} else {
			output, err = marshalScoreResult(Score(mapped))
		}
		if err != nil {
			return nil, err
		}
		self.Observations = append(self.Observations, calibrationReplayObservation{ImageId: image,
			ContainerId: fmt.Sprintf("%064x", 100+self.ReplayCalls), Owner: fmt.Sprintf("%064x", 200+self.ReplayCalls),
			InspectionSha256: heldOutDigest([]byte("synthetic inspection")), TerminalSha256: heldOutDigest([]byte("synthetic terminal")),
			ContainerCleanupVerified: true})
		return output, nil
	}
}
