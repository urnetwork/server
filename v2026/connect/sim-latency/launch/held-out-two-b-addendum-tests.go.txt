// Every object is visibly synthetic and owned by this fixture. The declaration
// cannot make these placeholder A/bridge files genuine accepted measurements.
func heldOutTwoBFixture(t *testing.T) (heldOutPreparationConfig, heldOutTwoBAddendum, []heldOutReference) {
	t.Helper()
	root := newCalibrationFileRoot(t)
	pin := func(name string) heldOutPinnedFile {
		return heldOutOpWrite(t, filepath.Join(root, name), []byte("synthetic "+name))
	}
	config := heldOutPreparationConfig{Protocol: heldOutProtocolThreeABridgeTwoB, PrivateRoot: filepath.Join(root, "private"), PublicPlanPath: filepath.Join(root, "public-plan.json"),
		Amendment:    &heldOutAmendmentPins{Approval: pin("original-approval"), OriginalPlan: pin("original-plan"), OriginalBaseSha: strings.Repeat("a", 40), OriginalImageId: "sha256:" + strings.Repeat("b", 64), Release: pin("release"), MarginFreeze: pin("margin-freeze"), BridgePlan: pin("bridge-plan"), BridgePublication: pin("bridge-publication"), BridgeInputs: pin("bridge-inputs"), BridgeReceipt: pin("bridge-receipt")},
		TwoBAddendum: &heldOutTwoBAddendumPins{Approval: pin("two-b-approval")}, StageAReport: pin("stage-a-report"), StageAInputs: pin("stage-a-inputs"), SourceLock: pin("source-lock"), References: pin("references"), Exclusions: pin("exclusions")}
	a := config.Amendment
	declaration := heldOutTwoBAddendum{Schema: 1, Kind: "sim-latency-held-out-two-b-addendum", Protocol: config.Protocol, BaseProtocol: calibrationProtocolThreeABridgeThreeB, PublishedAt: time.Date(2026, 10, 27, 0, 0, 0, 0, time.UTC),
		ApprovalSha256: config.TwoBAddendum.Approval.Sha256, OriginalApprovalSha256: a.Approval.Sha256, OriginalPlanSha256: a.OriginalPlan.Sha256, StageAReportSha256: config.StageAReport.Sha256, StageAInputsSha256: config.StageAInputs.Sha256, MarginFreezeSha256: a.MarginFreeze.Sha256,
		BridgePlanSha256: a.BridgePlan.Sha256, BridgePublicationSha256: a.BridgePublication.Sha256, BridgeInputsSha256: a.BridgeInputs.Sha256, BridgeReceiptSha256: a.BridgeReceipt.Sha256, ReleaseSha256: a.Release.Sha256, SourceLockSha256: config.SourceLock.Sha256, ReferencesSha256: config.References.Sha256, ExclusionsSha256: config.Exclusions.Sha256,
		Margin: .12, SeedCount: 2, RequiredOrderedSeeds: 2, ControlReplicates: 9, CandidateReplicates: 9, ExecutionLimitSeconds: 10800, FreshSeedsRequired: true, AllOutcomesRetained: true, NoReplacement: true}
	references := []heldOutReference{}
	for _, role := range []string{"no-op", "better", "worse"} {
		reference := heldOutReference{Role: role, Patch: pin(role + "-patch"), Direction: pin(role + "-direction")}
		references = append(references, reference)
		declaration.ReferenceTransfers = append(declaration.ReferenceTransfers, heldOutTwoBReferenceTransfer{Role: role, PatchSha256: reference.Patch.Sha256, OriginalDirectionSha256: reference.Direction.Sha256, ExpectedDirection: map[string]string{"no-op": "no-op", "better": "faster", "worse": "slower"}[role], Claim: "expected-direction-hypothesis-not-new-c2-measurement", FreshBridgeRequired: true, AllTwoOrderedBRequired: true})
	}
	config.TwoBAddendum.Declaration = heldOutOpJson(t, filepath.Join(root, "two-b-declaration.json"), declaration)
	return config, declaration, references
}

// A short observed list never selects an amendment. Existing five/four and
// three/all settings retain every other byte/setting and their own selectors.
func TestHeldOutTwoBAddendumIsNamedAndPreservesOldSettings(t *testing.T) {
	legacy, _ := heldOutSettingsForProtocol("")
	three, _ := heldOutSettingsForProtocol(calibrationProtocolThreeABridgeThreeB)
	two, err := heldOutSettingsForProtocol(heldOutProtocolThreeABridgeTwoB)
	if err != nil || two.Workload.SeedCount != 2 {
		t.Fatal(err)
	}
	if legacy.Workload.SeedCount != 5 || three.Workload.SeedCount != 3 {
		t.Fatal("existing cardinality changed")
	}
	two.Workload.SeedCount = 5
	three.Workload.SeedCount = 5
	if !reflect.DeepEqual(two, legacy) || !reflect.DeepEqual(three, legacy) {
		t.Fatal("new count altered another frozen setting")
	}
	config, _, _ := heldOutTwoBFixture(t)
	selected, _ := heldOutSettingsForProtocol(config.Protocol)
	if err := validateHeldOutProtocolSettings(config, selected); err != nil {
		t.Fatal(err)
	}
	config.TwoBAddendum = nil
	if err := validateHeldOutProtocolSettings(config, selected); err == nil {
		t.Fatal("two-B selected without independent addendum")
	}
	config.Protocol = calibrationProtocolThreeABridgeThreeB
	if err := validateHeldOutProtocolSettings(config, selected); err == nil {
		t.Fatal("two seeds reinterpreted as original three-B")
	}
}

// Both fresh seeds and every role must complete. No two-of-three selection,
// dropped failure, nonplaceable no-op or takeover substitution can be accepted.
func TestHeldOutTwoBAddendumRequiresBothCompleteOrderedSeeds(t *testing.T) {
	rounds, outcomes := newHeldOutAmendedOutcomes(2)
	accepted, ordered, reasons := acceptHeldOutOutcomesForProtocol(heldOutProtocolThreeABridgeTwoB, rounds, outcomes)
	if !accepted || ordered != 2 || len(reasons) != 0 {
		t.Fatal(accepted, ordered, reasons)
	}
	for _, mutate := range []func([]heldOutJobOutcome){
		func(outcomes []heldOutJobOutcome) { outcomes[5].RawScore = 25 },
		func(outcomes []heldOutJobOutcome) {
			outcomes[5].Complete = false
			outcomes[5].FailureCode = "synthetic_failure"
		},
		func(outcomes []heldOutJobOutcome) { outcomes[1].Placeable = false },
		func(outcomes []heldOutJobOutcome) { outcomes[1].TakeoverEligible = true },
	} {
		changed := append([]heldOutJobOutcome{}, outcomes...)
		mutate(changed)
		if accepted, _, _ := acceptHeldOutOutcomesForProtocol(heldOutProtocolThreeABridgeTwoB, rounds, changed); accepted {
			t.Fatal("failed fixed two-B screen accepted")
		}
	}
	for _, protocol := range []string{"", calibrationProtocolThreeABridgeThreeB} {
		if accepted, _, _ := acceptHeldOutOutcomesForProtocol(protocol, rounds, outcomes); accepted {
			t.Fatal("two seeds selected by old protocol", protocol)
		}
	}
	three, threeOutcomes := newHeldOutAmendedOutcomes(3)
	if accepted, _, _ := acceptHeldOutOutcomesForProtocol(heldOutProtocolThreeABridgeTwoB, three, threeOutcomes); accepted {
		t.Fatal("third seed silently discarded")
	}
	if accepted, _, _ := acceptHeldOutOutcomesForProtocol(heldOutProtocolThreeABridgeTwoB, rounds, outcomes[:5]); accepted {
		t.Fatal("missing reference outcome silently discarded")
	}
}

// The supplement pins unchanged old A and the bridge; malformed or changed
// numerical/policy facts fail before it can be used by the preparation caller.
func TestHeldOutTwoBAddendumPinsAllOriginalFacts(t *testing.T) {
	config, declaration, _ := heldOutTwoBFixture(t)
	margin := declaration.Margin
	if err := validateHeldOutTwoBAddendum(declaration, config, &margin); err != nil {
		t.Fatal(err)
	}
	content, _ := json.Marshal(declaration)
	for _, mutate := range []func(*heldOutTwoBAddendum){
		func(d *heldOutTwoBAddendum) { d.Protocol = calibrationProtocolThreeABridgeThreeB }, func(d *heldOutTwoBAddendum) { d.BaseProtocol = "" }, func(d *heldOutTwoBAddendum) { d.ApprovalSha256 = strings.Repeat("f", 64) },
		func(d *heldOutTwoBAddendum) { d.StageAReportSha256 = strings.Repeat("f", 64) }, func(d *heldOutTwoBAddendum) { d.BridgeReceiptSha256 = "" }, func(d *heldOutTwoBAddendum) { d.Margin += .01 },
		func(d *heldOutTwoBAddendum) { d.ControlReplicates = 8 }, func(d *heldOutTwoBAddendum) { d.ExecutionLimitSeconds++ }, func(d *heldOutTwoBAddendum) { d.RequiredOrderedSeeds = 1 }, func(d *heldOutTwoBAddendum) { d.SeedCount = 3 },
		func(d *heldOutTwoBAddendum) { d.NoReplacement = false }, func(d *heldOutTwoBAddendum) { d.FreshSeedsRequired = false }, func(d *heldOutTwoBAddendum) { d.ReferenceTransfers[0].AllTwoOrderedBRequired = false }, func(d *heldOutTwoBAddendum) { d.ReferenceTransfers[1].Role = d.ReferenceTransfers[0].Role },
	} {
		var changed heldOutTwoBAddendum
		if err := json.Unmarshal(content, &changed); err != nil {
			t.Fatal(err)
		}
		mutate(&changed)
		if err := validateHeldOutTwoBAddendum(changed, config, &margin); err == nil {
			t.Fatal("changed two-B fact accepted")
		}
	}
	base, err := heldOutBaseProvenanceConfiguration(config)
	if err != nil || base.Protocol != calibrationProtocolThreeABridgeThreeB || base.TwoBAddendum != nil || base.StageAReport != config.StageAReport || base.StageAInputs != config.StageAInputs || !reflect.DeepEqual(base.Amendment, config.Amendment) {
		t.Fatal("old A/bridge metadata was relabeled", err)
	}
}

// The old encoder is retained only as a deterministic byte-level regression
// oracle. Production never runs it to relabel a new protocol or commitment.
func heldOutPreTwoBCommitment(t *testing.T, seedHex string, round heldOutPublicRound, plan heldOutPrivatePlan, release heldOutRelease) string {
	t.Helper()
	seed, err := hex.DecodeString(seedHex)
	if err != nil {
		t.Fatal(err)
	}
	round.SeedCommitment = ""
	protocol, approval, originalPlan, marginFreeze, bridgePlan, bridgeReceipt, newRelease := "", "", "", "", "", "", ""
	if plan.Config.Protocol != "" {
		pins := plan.Config.Amendment
		protocol, approval, originalPlan, marginFreeze, bridgePlan, bridgeReceipt, newRelease = plan.Config.Protocol, pins.Approval.Sha256, pins.OriginalPlan.Sha256, pins.MarginFreeze.Sha256, pins.BridgePlan.Sha256, pins.BridgeReceipt.Sha256, pins.Release.Sha256
	}
	content, err := json.Marshal(struct {
		Domain                                          string             `json:"domain"`
		CampaignId                                      string             `json:"campaign_id"`
		Round                                           heldOutPublicRound `json:"round"`
		Source, Image, Lock, Policy, Report, References string
		Margin                                          float64 `json:"margin"`
		Protocol                                        string  `json:"protocol,omitempty"`
		Approval                                        string  `json:"protocol_approval_sha256,omitempty"`
		OriginalPlan                                    string  `json:"original_plan_sha256,omitempty"`
		MarginFreeze                                    string  `json:"margin_freeze_sha256,omitempty"`
		BridgePlan                                      string  `json:"bridge_plan_sha256,omitempty"`
		BridgeReceipt                                   string  `json:"bridge_receipt_sha256,omitempty"`
		NewRelease                                      string  `json:"new_release_sha256,omitempty"`
	}{Domain: "urnetwork-held-out-stage-b-v1", CampaignId: plan.CampaignId, Round: round, Source: release.BaseSha, Image: release.ImageId, Lock: release.SourceLockSha256, Policy: plan.PolicySha256, Report: plan.Config.StageAReport.Sha256, References: plan.Config.References.Sha256, Margin: plan.Margin, Protocol: protocol, Approval: approval, OriginalPlan: originalPlan, MarginFreeze: marginFreeze, BridgePlan: bridgePlan, BridgeReceipt: bridgeReceipt, NewRelease: newRelease})
	if err != nil {
		t.Fatal(err)
	}
	hash := sha256.New()
	hash.Write([]byte("urnetwork-held-out-stage-b-seed-v1\x00"))
	hash.Write(seed)
	hash.Write(content)
	return hex.EncodeToString(hash.Sum(nil))
}

// Existing protocol commitments encode identically. New public commitments
// bind the separately approved addendum as well as unchanged base provenance.
func TestHeldOutTwoBAddendumCommitmentAndPublicPlanBindings(t *testing.T) {
	config, _, _ := heldOutTwoBFixture(t)
	base, _ := heldOutBaseProvenanceConfiguration(config)
	release := frozenHeldOutRelease()
	round := heldOutPublicRound{Position: 1, RoundId: heldOutOpId(71)}
	plan := heldOutPrivatePlan{CampaignId: heldOutOpId(70), Config: base, Margin: .12, PolicySha256: strings.Repeat("a", 64)}
	seed := strings.Repeat("1", 64)
	for _, protocol := range []string{"", calibrationProtocolThreeABridgeThreeB} {
		old := plan
		old.Config.Protocol = protocol
		if protocol == "" {
			old.Config.Amendment = nil
		}
		actual, err := heldOutRoundCommitment(seed, round, old, release)
		if err != nil || actual != heldOutPreTwoBCommitment(t, seed, round, old, release) {
			t.Fatal("old commitment bytes changed", protocol, err)
		}
	}
	plan.Config = config
	first, err := heldOutRoundCommitment(seed, round, plan, release)
	if err != nil {
		t.Fatal(err)
	}
	public := publicHeldOutPlan(plan, strings.Repeat("b", 64), release)
	if public.Protocol != config.Protocol || public.ProtocolApprovalSha256 != config.TwoBAddendum.Approval.Sha256 || public.RequiredOrderedSeeds != 2 || public.BaseProtocol != calibrationProtocolThreeABridgeThreeB || public.BaseProtocolApprovalSha256 != config.Amendment.Approval.Sha256 || public.TwoBAddendumSha256 != config.TwoBAddendum.Declaration.Sha256 || public.ControlReplicates != 9 || public.CandidateReplicates != 9 || public.ExecutionLimitSeconds != 10800 {
		t.Fatal("new publication omitted its exact scope/pins")
	}
	changed := plan
	addendum := *config.TwoBAddendum
	changed.Config.TwoBAddendum = &addendum
	changed.Config.TwoBAddendum.Declaration.Sha256 = strings.Repeat("f", 64)
	if next, err := heldOutRoundCommitment(seed, round, changed, release); err == nil && next == first {
		t.Fatal("new approval/declaration was not commitment bound")
	}
	encoded, _ := json.Marshal(publicHeldOutPlan(heldOutPrivatePlan{Config: base}, "", release))
	if bytes.Contains(encoded, []byte("two_b_")) || bytes.Contains(encoded, []byte("base_protocol")) {
		t.Fatal("old public plan acquired new amendment fields")
	}
}

// Reveals only the two originally committed synthetic seeds after an explicit
// closure, with new approval metadata; altering the addendum changes the hash.
func TestHeldOutTwoBAddendumRevealBindsOriginalTwoSeeds(t *testing.T) {
	config, declaration, _ := heldOutTwoBFixture(t)
	settings, _ := heldOutSettingsForProtocol(config.Protocol)
	release := frozenHeldOutRelease()
	rounds, outcomes := newHeldOutAmendedOutcomes(2)
	plan := heldOutPrivatePlan{CampaignId: heldOutOpId(80), PreparedAt: declaration.PublishedAt.Add(time.Minute), Config: config, Margin: declaration.Margin, PolicySha256: strings.Repeat("a", 64), Rounds: rounds}
	for index := range plan.Rounds {
		plan.Rounds[index].PrivateSeedHex = strings.Repeat(string(rune('1'+index)), 64)
		digest, err := heldOutRoundCommitment(plan.Rounds[index].PrivateSeedHex, plan.Rounds[index].Public, plan, release)
		if err != nil {
			t.Fatal(err)
		}
		plan.Rounds[index].Public.SeedCommitment = digest
	}
	publicPin := heldOutPinnedFile{Path: config.PublicPlanPath, Sha256: strings.Repeat("b", 64)}
	closed := plan.PreparedAt.Add(time.Minute)
	result := heldOutCampaignResult{Schema: 1, Kind: "sim-latency-held-out-screen", CampaignId: plan.CampaignId, PlanSha256: publicPin.Sha256, Complete: true, Accepted: true, OrderedSeeds: 2, Reasons: []string{}, Outcomes: outcomes, ClosedAt: closed}
	resultPin := heldOutOpJson(t, filepath.Join(config.PrivateRoot, "campaign-result.json"), result)
	closure := heldOutClosure{Schema: 1, Kind: "sim-latency-held-out-campaign-closure", CampaignId: plan.CampaignId, PublicPlanSha256: publicPin.Sha256, ResultSha256: resultPin.Sha256, ClosedAt: closed, NoFurtherEvaluations: true, Reason: "synthetic closed screen", Evidence: []heldOutPinnedFile{config.TwoBAddendum.Declaration}}
	reveal := heldOutRevealConfig{Schema: 1, Kind: "sim-latency-held-out-reveal", Execution: heldOutExecutionConfig{PublicPlan: publicPin}, Result: resultPin, Closure: heldOutOpJson(t, filepath.Join(filepath.Dir(config.PrivateRoot), "closure.json"), closure), RevealPath: filepath.Join(filepath.Dir(config.PrivateRoot), "reveal.json")}
	if _, err := revealHeldOutCampaign(reveal, plan, release, settings); err != nil {
		t.Fatal(err)
	}
	content, err := os.ReadFile(reveal.RevealPath)
	if err != nil {
		t.Fatal(err)
	}
	for _, round := range plan.Rounds {
		if !bytes.Contains(content, []byte(round.PrivateSeedHex)) {
			t.Fatal("original committed seed omitted")
		}
	}
	if !bytes.Contains(content, []byte(config.TwoBAddendum.Declaration.Sha256)) || !bytes.Contains(content, []byte(config.Protocol)) {
		t.Fatal("reveal omitted its separately approved two-B scope")
	}
	changed := plan
	addendum := *config.TwoBAddendum
	changed.Config.TwoBAddendum = &addendum
	changed.Config.TwoBAddendum.Declaration.Sha256 = strings.Repeat("f", 64)
	reveal.RevealPath += ".changed"
	if _, err := revealHeldOutCampaign(reveal, changed, release, settings); err == nil {
		t.Fatal("revealed seeds under a changed addendum")
	}
}
