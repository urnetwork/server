// Changed seed counts require an explicit separately named protocol.
func TestHeldOutAmendmentKeepsFiveSeedDefault(t *testing.T) {
	legacy, err := heldOutSettingsForProtocol("")
	if err != nil || !reflect.DeepEqual(legacy, defaultHeldOutOperatorSettings()) {
		t.Fatalf("historical settings changed: %v", err)
	}
	amended, err := heldOutSettingsForProtocol(calibrationProtocolThreeABridgeThreeB)
	if err != nil || amended.Workload.SeedCount != 3 {
		t.Fatalf("named amendment rejected: %v", err)
	}
	amended.Workload.SeedCount = 5
	if !reflect.DeepEqual(amended, legacy) {
		t.Fatal("non-cardinality held-out settings changed")
	}
	if _, err := heldOutSettingsForProtocol("synthetic-unapproved-protocol"); err == nil {
		t.Fatal("unapproved protocol accepted")
	}
	if err := validateHeldOutProtocolSettings(heldOutPreparationConfig{}, amended); err != nil {
		t.Fatal(err)
	}
	amended.Workload.SeedCount = 3
	if err := validateHeldOutProtocolSettings(heldOutPreparationConfig{}, amended); err == nil {
		t.Fatal("cardinality silently selected an amendment")
	}
}

// Builds fixed healthy ordered outcomes without entropy, native generation,
// process hooks, network services or an executable candidate.
func newHeldOutAmendedOutcomes(count int) ([]heldOutPrivateRound, []heldOutJobOutcome) {
	rounds := []heldOutPrivateRound{}
	outcomes := []heldOutJobOutcome{}
	for index := 0; index < count; index++ {
		round := heldOutPrivateRound{Public: heldOutPublicRound{Position: index + 1, RoundId: fmt.Sprintf("synthetic-round-%d", index), Order: [3]string{"worse", "no-op", "better"}, Jobs: []heldOutPlannedJob{}}}
		for jobIndex, role := range round.Public.Order {
			job := heldOutPlannedJob{Position: jobIndex + 1, Role: role, JobId: fmt.Sprintf("synthetic-job-%d-%d", index, jobIndex)}
			round.Public.Jobs = append(round.Public.Jobs, job)
			pValue := 0.5
			outcomes = append(outcomes, heldOutJobOutcome{JobId: job.JobId, RoundId: round.Public.RoundId, Role: role, Complete: true, RawScore: map[string]float64{"better": 10, "no-op": 20, "worse": 30}[role], Placeable: true, Significance: &ScoreSignificance{OneSidedPValue: &pValue}})
		}
		rounds = append(rounds, round)
	}
	return rounds, outcomes
}

// Every amended seed must order correctly; two successes cannot replace three.
func TestHeldOutAmendmentRequiresAllThreeOrderedSeeds(t *testing.T) {
	rounds, outcomes := newHeldOutAmendedOutcomes(3)
	accepted, ordered, reasons := acceptHeldOutOutcomesForProtocol(calibrationProtocolThreeABridgeThreeB, rounds, outcomes)
	if !accepted || ordered != 3 || len(reasons) != 0 {
		t.Fatalf("complete amended ordering rejected: %d %v", ordered, reasons)
	}
	outcomes[8].RawScore = 25
	accepted, ordered, reasons = acceptHeldOutOutcomesForProtocol(calibrationProtocolThreeABridgeThreeB, rounds, outcomes)
	if accepted || ordered != 2 || !reflect.DeepEqual(reasons, []string{"not_all_three_correctly_ordered_seeds"}) {
		t.Fatalf("two of three accepted: %d %v", ordered, reasons)
	}
}

// The published historical four-of-five rule remains usable only as itself.
func TestHeldOutAmendmentDoesNotRewriteFiveSeedHistory(t *testing.T) {
	rounds, outcomes := newHeldOutAmendedOutcomes(5)
	outcomes[14].RawScore = 25
	accepted, ordered, reasons := acceptHeldOutOutcomes(rounds, outcomes)
	if !accepted || ordered != 4 || len(reasons) != 0 {
		t.Fatalf("historical four-of-five changed: %d %v", ordered, reasons)
	}
	if accepted, _, _ := acceptHeldOutOutcomesForProtocol(calibrationProtocolThreeABridgeThreeB, rounds, outcomes); accepted {
		t.Fatal("historical five seeds relabeled as amended screen")
	}
	rounds, outcomes = newHeldOutAmendedOutcomes(3)
	if accepted, _, _ := acceptHeldOutOutcomes(rounds, outcomes); accepted {
		t.Fatal("three seeds implicitly selected amendment")
	}
}

// All no-ops must remain placeable without takeover regardless of ordering.
func TestHeldOutAmendmentRetainsAllNoopGates(t *testing.T) {
	for _, takeover := range []bool{false, true} {
		rounds, outcomes := newHeldOutAmendedOutcomes(3)
		if takeover {
			outcomes[4].TakeoverEligible = true
		} else {
			outcomes[4].Placeable = false
		}

			t.Fatalf("no-op failure discarded; takeover=%t ordered=%d", takeover, ordered)
		}
	}
}

			t.Errorf("reference count %d accepted", count)
		}
	}
}

// Synthetic source text contains precisely the four required old anchors and
// an unrelated policy sentinel that the prospective transformation preserves.
func TestHeldOutAmendmentSourceTransformIsExactlyFourHunks(t *testing.T) {
	original := []byte("// synthetic unrelated policy sentinel\n" +
		"type evaluatorSourceLock struct {\n\tSchema              int               `json:\"schema\"`\n\tDevelopmentSnapshot bool              `json:\"development_snapshot\"`\n\tRepositories        map[string]string `json:\"repositories\"`\n}\n" +
		"\tif lock.DevelopmentSnapshot {\n\t\treturn nil, errors.New(\"development snapshot cannot be used for an epoch-bound evaluation\")\n\t}\n\trepositoryNames := sourceRepositoryNamesForMode(staging)\n" +
		"\t\tif repository != \"server\" {\n\t\t\treturn fmt.Errorf(\"evaluator repository %s head does not match its authenticated source lock\", repository)\n" +

	if err != nil {
		t.Fatal(err)
	}

		t.Fatal("already changed source accepted as a pristine C1 transformation")
	}
	duplicated := append(append([]byte{}, original...), []byte("\t\t\treturn errors.New(\"candidate server commit is not exactly one commit above its configured epoch base\")")...)
	if _, err := heldOutProspectiveSourceEpoch(duplicated); err == nil {

	}

// Direct tree checks reject non-child commits and unrelated changed files
// before any source object, executable or candidate can be consumed.
func TestHeldOutAmendmentInstrumentRejectsUnrelatedTreeChanges(t *testing.T) {
	for _, badParent := range []bool{false, true} {
		descriptor := heldOutAmendedRelease{Repository: "/synthetic/source", BaseSha: strings.Repeat("b", 40)}
		old := heldOutRelease{BaseSha: strings.Repeat("a", 40)}
		calls := 0
		err := authenticateHeldOutInstrumentSource(context.Background(), descriptor, old, defaultHeldOutOperatorSettings(), func(_ context.Context, program string, args ...string) ([]byte, error) {
			calls++
			joined := strings.Join(args, " ")
			if strings.Contains(joined, "rev-list") {
				if badParent {
					return []byte(descriptor.BaseSha + " " + strings.Repeat("c", 40)), nil
				}
				return []byte(descriptor.BaseSha + " " + old.BaseSha), nil
			}
			if strings.Contains(joined, "diff-tree") {
				return []byte("connect/sim-latency/score.go\nconnect/sim-latency/source_epoch.go\n"), nil
			}
			t.Fatal("unapproved tree reached source/candidate consumption")
			return nil, nil
		})
		if err == nil || calls < 1 || calls > 2 {
			t.Fatalf("unapproved tree accepted; badParent=%t calls=%d err=%v", badParent, calls, err)
		}
	}
}

// The original whole invocation, including cleanup, remains capped at 10800.
func TestHeldOutBridgeRequiresOriginalBoundedTerminal(t *testing.T) {
	start := time.Unix(1600000000, 0).UTC()
	valid := func(duration int, cleanup bool) []byte {
		marker := ""
		if cleanup {
			marker = "terminal_cleanup_verified\n"
		}
		return []byte(fmt.Sprintf("start_utc=%s\nend_utc=%s\nevaluator_exit=0 duration_seconds=%d\n%s", start.Format(time.RFC3339), start.Add(time.Duration(duration)*time.Second).Format(time.RFC3339), duration, marker))
	}
	first, last, err := heldOutBridgeInterval(valid(10800, true), 10800)
	if err != nil || first != start || last.Sub(first) != 10800*time.Second {
		t.Fatalf("original deadline boundary rejected: %v", err)
	}
	for _, content := range [][]byte{valid(10801, true), valid(100, false), append(valid(100, true), []byte("start_utc="+start.Format(time.RFC3339)+"\n")...)} {
		if _, _, err := heldOutBridgeInterval(content, 10800); err == nil {
			t.Fatal("reset/over-budget/incomplete terminal accepted")
		}
	}
}

	request := heldOutRequest{PatchSha256: strings.Repeat("2", 64), ProvidersSha256: strings.Repeat("3", 64)}
	bindings := []calibrationArtifactBinding{}
	pins := map[string]heldOutPinnedFile{}
	for index := 0; index < 465; index++ {
		name := fmt.Sprintf("evidence/synthetic-%03d.json", index)
		if index < 5 {
			name = []string{"accounting.json", "baseline.json", "resources.json", "score.json", "evaluation-progress.json"}[index]
		}
		content := []byte("{}\n")
		if index == 4 {
			content = []byte(`{"phase":"complete","baseline_completed":9,"candidate_completed":9}`)
		}
		pin := heldOutOpWrite(t, filepath.Join(root, name), content)
		pins[name] = pin
		bindings = append(bindings, calibrationArtifactBinding{Path: name, Sha256: pin.Sha256, Bytes: int64(len(content))})
	}
	evidence := heldOutOpJson(t, filepath.Join(root, "evidence-manifest.json"), map[string]any{"artifacts": bindings})
	input.EvidenceManifestSha256 = evidence.Sha256
	pins["evidence-manifest.json"] = evidence
	candidate := "sha256:" + strings.Repeat("4", 64)
	complete := heldOutCompletion{Schema: 1, Kind: "sim-latency-worker-evaluation-complete", JobId: input.JobId, RoundId: input.RoundId, Attempt: 1, BaseImageId: release.ImageId, CandidateImageId: &candidate, PatchSha256: request.PatchSha256, ProvidersSha256: request.ProvidersSha256, CleanupComplete: true, Artifacts: map[string]string{}}
	for key, path := range map[string]string{"accounting": "accounting.json", "baseline": "baseline.json", "resources": "resources.json", "score": "score.json", "evidence_manifest": "evidence-manifest.json"} {
		complete.Artifacts[key] = pins[path].Sha256
	}
	heldOutOpJson(t, filepath.Join(root, "evaluation.complete.json"), complete)
	return input, release, request, settings
}

// Fixed map/cardinality claims do not replace authenticating every raw byte.
func TestHeldOutBridgeAuthenticatesWholeNativeCensus(t *testing.T) {
	input, release, request, settings := newHeldOutBridgeCensusFixture(t)
	if err := authenticateHeldOutBridgeCensus(input, release, request, settings); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(filepath.Join(input.ArtifactDirectory, "evidence", "synthetic-464.json"), []byte("changed synthetic bytes"), 0600); err != nil {
		t.Fatal(err)
	}
	if err := authenticateHeldOutBridgeCensus(input, release, request, settings); err == nil {
		t.Fatal("self-reported census hid changed raw evidence")
	}
}

// Completion artifacts are a keyed map, not an array or a partial green flag.
func TestHeldOutBridgeRejectsChangedCompletionMapCoverage(t *testing.T) {
	input, release, request, settings := newHeldOutBridgeCensusFixture(t)
	path := filepath.Join(input.ArtifactDirectory, "evaluation.complete.json")
	content, err := os.ReadFile(path)
	if err != nil {
		t.Fatal(err)
	}
	var complete heldOutCompletion
	if err := json.Unmarshal(content, &complete); err != nil {
		t.Fatal(err)
	}
	delete(complete.Artifacts, "resources")
	heldOutOpJson(t, path, complete)
	if err := authenticateHeldOutBridgeCensus(input, release, request, settings); err == nil {
		t.Fatal("four-item completion map accepted as five native artifact bindings")
	}
}
