// Creates separate retained old input/output namespaces and public metadata.
func (self *heldOutFreshSingleFixture) configureLineage(t *testing.T) {
	t.Helper()
	op := self.Base.Preparation
	root := filepath.Join(op.Root, "synthetic-old-campaign")
	rounds, outcomes := newHeldOutAmendedOutcomes(2)
	for _, round := range rounds {
		for _, job := range round.Public.Jobs {
			artifacts := filepath.Join(heldOutJobRoot(root, round.Public, job), "artifacts")
			for _, name := range []string{"canonical.patch", "precommitted-request.json", "providers.yml"} {
				heldOutOpWrite(t, filepath.Join(artifacts, name), []byte("synthetic immutable original input "+name+"\n"))
			}
		}
	}
	originalPrivate := heldOutPrivatePlan{Schema: 1, Kind: "sim-latency-held-out-private-plan",
		CampaignId: heldOutOpId(9401), PreparedAt: time.Unix(1500000000, 0).UTC(), Rounds: rounds}
	for index := range originalPrivate.Rounds {
		round := &originalPrivate.Rounds[index]
		round.PrivateSeedHex = fmt.Sprintf("%064x", 9101+index)
		seed, err := heldOutEffectiveSeed(round.PrivateSeedHex)
		if err != nil {
			t.Fatal(err)
		}
		round.EffectiveSeed = seed
		round.Public.ProvidersSha256 = heldOutDigest([]byte(fmt.Sprintf("synthetic old providers %d", index)))
		round.Public.SeedCommitment = heldOutDigest([]byte(fmt.Sprintf("synthetic old commitment %d", index)))
	}
	self.Facts.OriginalRoot = root
	self.Facts.OriginalPrivatePlan = heldOutOpJson(t, filepath.Join(root, "private-plan.json"), originalPrivate)
	public := heldOutPublicPlan{Schema: 1, Kind: "sim-latency-held-out-public-plan", Protocol: heldOutProtocolThreeABridgeTwoB,
		CampaignId: originalPrivate.CampaignId, PrivatePlanSha256: self.Facts.OriginalPrivatePlan.Sha256,
		DeferredFields: []string{"round_baseline_path", "round_baseline_sha256"}, Rounds: []heldOutPublicRound{}}
	for _, round := range originalPrivate.Rounds {
		public.Rounds = append(public.Rounds, round.Public)
	}
	self.Facts.OriginalPublicPlan = heldOutOpJson(t, filepath.Join(op.Root, "synthetic-old-public-plan.json"), public)
	self.Facts.SelectedPublicPlan = heldOutOpWrite(t, filepath.Join(op.Root, "synthetic-old-selected-public.json"),
		[]byte("{\"synthetic\":\"immutable selected public lineage\"}\n"))
	for index := range outcomes[:3] {
		if outcomes[index].Role == "worse" {
			outcomes[index].RawScore = 15
		}
	}
	failed := heldOutCampaignResult{Schema: 1, Kind: "sim-latency-held-out-screen", CampaignId: public.CampaignId,
		PlanSha256: self.Facts.SelectedPublicPlan.Sha256, Complete: true, Accepted: false, OrderedSeeds: 0,
		Reasons: []string{"first_committed_seed_not_correctly_ordered"}, Outcomes: outcomes[:3], ClosedAt: time.Unix(1600000045, 0).UTC()}
	self.Facts.FailedResult = heldOutOpJson(t, filepath.Join(root, "campaign-result.json"), failed)
	self.Declaration.OriginalExecutionStarted = heldOutOpJson(t, filepath.Join(root, "execution.started.json"),
		map[string]any{"schema": 1, "synthetic": "spent original campaign, never reset"})
	content, err := readHeldOutPinned(op.Config.Exclusions, op.Settings.MaximumBytes, false)
	if err != nil || json.Unmarshal(content, &self.Exclusions) != nil {
		t.Fatal("synthetic exclusions unavailable", err)
	}
	self.Exclusions.Sources = append(self.Exclusions.Sources, self.Facts.OriginalPrivatePlan)
	for _, round := range originalPrivate.Rounds {
		self.Exclusions.PrivateSeedHexes = append(self.Exclusions.PrivateSeedHexes, round.PrivateSeedHex)
		self.Exclusions.EffectiveSeeds = append(self.Exclusions.EffectiveSeeds, round.EffectiveSeed)
		self.Exclusions.ProviderDigests = append(self.Exclusions.ProviderDigests, round.Public.ProvidersSha256)
		self.Exclusions.RetiredCommitments = append(self.Exclusions.RetiredCommitments, round.Public.SeedCommitment)
	}
}

// Every new policy-bound native build has actual typed original-source/builder
// sidecars, while better/no-op preserve their original synthetic patch bytes.
func (self *heldOutFreshSingleFixture) configureReferences(t *testing.T) {
	t.Helper()
	op := self.Base.Preparation
	var lock heldOutSourceLock
	content, err := readHeldOutPinned(op.Config.SourceLock, op.Settings.MaximumBytes, false)
	if err != nil || json.Unmarshal(content, &lock) != nil {
		t.Fatal("synthetic lock unavailable", err)
	}
	var manifest heldOutRuntimeManifest
	if err := json.Unmarshal(self.Base.RuntimeContentBytesKVs[op.Release.Runtime.Manifest.Path], &manifest); err != nil {
		t.Fatal(err)
	}
	policyBytes, err := heldOutFreshSinglePatchPolicyBytes(self.Template.PatchPolicy)
	if err != nil {
		t.Fatal(err)
	}
	policySha256 := heldOutDigest(policyBytes)
	self.Declaration.ReferencePolicySha256 = policySha256
	self.Declaration.ReferencePolicy = heldOutOpWrite(t, filepath.Join(op.Root, "inputs", "fresh-scored-patch-policy.json"), policyBytes)
	self.References = append([]heldOutReference{}, self.Base.References...)
	for index := range self.References {
		reference := &self.References[index]
		directory := filepath.Join(op.Root, "synthetic-fresh-reference", reference.Role)
		if reference.Role == "worse" {
			reference.Patch = heldOutOpWrite(t, filepath.Join(directory, "canonical.patch"), []byte("synthetic changed causal regression patch\n"))
		}
		oldBytes, err := readHeldOutPinned(reference.NativeBuild, op.Settings.MaximumBytes, false)
		if err != nil {
			t.Fatal(err)
		}
		var build heldOutNativeBuild
		if err := json.Unmarshal(oldBytes, &build); err != nil {
			t.Fatal(err)
		}
		source := heldOutRuntimeSource{Schema: 1, Kind: "sim-latency-evaluation-source", Temporary: true,
			BaseImageId: op.Release.ImageId, BaseSha: op.Release.BaseSha, Branch: "sim-latency",
			SourceLockSha256: op.Config.SourceLock.Sha256, CandidatePathBase: "mono_root_v1", Repositories: lock.Repositories}
		sourcePin := heldOutOpWrite(t, filepath.Join(directory, "evidence", "candidate-original-source.json"), heldOutRuntimeJson(t, source))
		builderPin := heldOutOpWrite(t, filepath.Join(directory, "evidence", "builder-identity.json"), heldOutRuntimeJson(t, manifest.Builder))
		reference.ImageId = "sha256:" + heldOutDigest([]byte("synthetic new-policy image "+reference.Role))
		build.ImageId, build.PatchSha256, build.PolicySha256 = reference.ImageId, reference.Patch.Sha256, policySha256
		build.OriginalSourceIdentitySha256 = sourcePin.Sha256
		build.ImageKey = heldOutDigest([]byte(strings.Join([]string{build.BaseImageId, build.PatchSha256, build.PolicySha256, build.BuilderSha256}, "\x00")))
		build.Image = "urnetwork/sim-latency-submission:" + build.ImageKey[:32]
		reference.NativeBuild = heldOutOpJson(t, filepath.Join(directory, "native-build.json"), build)
		for _, kind := range []string{"build", "correctness", "direction"} {
			receipt := heldOutReferenceQualification{Schema: 1, Kind: "sim-latency-held-out-reference-" + kind,
				Role: reference.Role, Accepted: true, BaseSha: op.Release.BaseSha, SourceLockSha256: op.Release.SourceLockSha256,
				BaseImageId: op.Release.ImageId, PatchSha256: reference.Patch.Sha256, CandidateSha: reference.CandidateSha,
				ImageId: reference.ImageId, BinarySha256: reference.BinarySha256, GoVersion: "go1.27.3",
				Evidence: []heldOutPinnedFile{sourcePin, builderPin}}
			if kind == "direction" {
				receipt.Direction = map[string]string{"no-op": "no-op", "better": "faster", "worse": "slower"}[reference.Role]
				if reference.Role == "worse" {
					receipt.Evidence = append(receipt.Evidence, self.Declaration.RegressionPilotPlan, self.Declaration.RegressionPilotPublication)
				}
			}
			pin := heldOutOpJson(t, filepath.Join(directory, kind+"-qualification.json"), receipt)
			switch kind {
			case "build":
				reference.Build = pin
			case "correctness":
				reference.Correctness = pin
			case "direction":
				reference.Direction = pin
			}
		}
	}
	op.Config.References = heldOutOpJson(t, filepath.Join(op.Root, "inputs", "fresh-references.json"), self.References)
}

// The producer's new first fence is exercised before any injected entropy read.
func (self *heldOutFreshSingleFixture) prepare(t *testing.T) (*heldOutPrivatePlan, error) {
	t.Helper()
	op := self.Base.Preparation
	inputs, err := self.authenticate()
	if err != nil {
		return nil, err
	}
	op.refreshConfig(t)
	public, err := prepareHeldOutCampaign(t.Context(), op.Config, heldOutDigest(op.ConfigBytes), *inputs, op.Release, op.Settings, op.Hooks, func() error {
		self.Rechecks++
		if self.BeforeRecheck != nil {
			self.BeforeRecheck()
		}
		again, err := self.authenticate()
		if err != nil {
			return err
		}
		if !reflect.DeepEqual(again, inputs) {
			return errors.New("synthetic fresh inputs changed")
		}
		return nil
	})
	if err != nil {
		return nil, err
	}
	op.PrivatePin = heldOutPinnedFile{Path: filepath.Join(op.Config.PrivateRoot, "private-plan.json"), Sha256: public.PrivatePlanSha256}
	content, err := os.ReadFile(op.Config.PublicPlanPath)
	if err != nil {
		return nil, err
	}
	op.PublicPin = heldOutPinnedFile{Path: op.Config.PublicPlanPath, Sha256: heldOutDigest(content)}
	return verifyHeldOutPlanWithContract(op.PrivatePin, op.PublicPin, op.Release, op.Settings, &self.Base.Contract)
}

// Process hooks are the only substitutions: real journals, deadlines, baseline
// authentication and the unchanged final acceptance reducer remain composed.
func (self *heldOutFreshSingleFixture) execute(t *testing.T, plan *heldOutPrivatePlan) (*heldOutCampaignResult, *heldOutExecFixture, error) {
	t.Helper()
	op := self.Base.Preparation
	verified, err := verifyHeldOutPlanWithContract(op.PrivatePin, op.PublicPin, op.Release, op.Settings, &self.Base.Contract)
	if err != nil || !reflect.DeepEqual(plan, verified) {
		return nil, nil, errors.Join(err, errors.New("synthetic fresh verification changed"))
	}
	fixture := &heldOutExecFixture{Preparation: op, Plan: verified, Clock: time.Now().UTC().Add(24 * time.Hour).Truncate(time.Second),
		Controls: map[string]*heldOutFrozenControl{}, Config: heldOutExecutionConfig{Schema: 1, Kind: "sim-latency-held-out-execution", PrivatePlan: op.PrivatePin, PublicPlan: op.PublicPin}}
	fixture.Config.Publication = heldOutOpJson(t, filepath.Join(op.Root, "synthetic-fresh-publication.json"), heldOutPublication{Schema: 1, Kind: "sim-latency-held-out-plan-publication",
		PlanSha256: op.PublicPin.Sha256, PublishedAt: plan.PreparedAt.Add(time.Second), Commit: strings.Repeat("c", 40), Repository: op.Root,
		PlanPath: "public/fresh-plan.json", RemoteRef: "refs/remotes/origin/synthetic", PulledPushed: true, Evidence: []heldOutPinnedFile{op.Config.FreshSingle.Declaration}})
	fixture.Hooks = heldOutExecutionHooks{Now: func() time.Time { return fixture.Clock },
		CheckHost: func(context.Context, heldOutPrivatePlan) (string, error) {
			return heldOutDigest([]byte("synthetic unchanged inventory")), nil
		},
		Invoke: func(ctx context.Context, root string, start heldOutJobStart) error {
			deadline, ok := ctx.Deadline()
			if !ok || !deadline.Equal(start.Deadline) || start.Deadline.Sub(start.StartedAt) != 3*time.Hour {
				return errors.New("synthetic fresh deadline changed")
			}
			fixture.Invocations = append(fixture.Invocations, start)
			fixture.Clock = fixture.Clock.Add(2 * time.Second)
			return writeHeldOutExclusive(filepath.Join(root, "manager-terminal.log"), []byte(fmt.Sprintf("start_utc=%s\ndeadline_utc=%s\nevaluator_exit=0 duration_seconds=2\nend_utc=%s\n", start.StartedAt.Format(time.RFC3339), start.Deadline.Format(time.RFC3339), fixture.Clock.Format(time.RFC3339))), 0600)
		},
		Authenticate: func(_ context.Context, _ heldOutPrivatePlan, round heldOutPrivateRound, job heldOutPlannedJob, _ heldOutRequest, control *heldOutFrozenControl) (heldOutJobOutcome, *heldOutFrozenControl, error) {
			if control == nil {
				control = newHeldOutExecControl(t, fixture, round, job)
				fixture.Controls[round.Public.RoundId] = control
			}
			pValue := .5
			outcome := heldOutJobOutcome{JobId: job.JobId, RoundId: round.Public.RoundId, Role: job.Role, Complete: true,
				RawScore: map[string]float64{"better": 8, "no-op": 10, "worse": 12}[job.Role], Placeable: true,
				Significance: &ScoreSignificance{OneSidedPValue: &pValue}, BaselineSha256: control.Baseline.Sha256}
			for index := 0; index < plan.Policy.Replicates; index++ {
				outcome.CandidateEvaluationIds = append(outcome.CandidateEvaluationIds, fmt.Sprintf("synthetic-fresh-%s-%02d", job.JobId, index))
			}
			return outcome, control, nil
		}}
	result, err := executeHeldOutCampaign(t.Context(), fixture.Config, *verified, op.Release, op.Settings, fixture.Hooks)
	return result, fixture, err
}

// Only the explicitly named protocol selects one draw; old defaults are identical.
func TestHeldOutFreshSingleCountDoesNotRewriteOldProtocols(t *testing.T) {
	legacy, _ := heldOutSettingsForProtocol("")
	for protocol, count := range map[string]int{"": 5, calibrationProtocolThreeABridgeThreeB: 3, heldOutProtocolThreeABridgeTwoB: 2, heldOutProtocolFreshSingleB: 1} {
		settings, err := heldOutSettingsForProtocol(protocol)
		if err != nil || settings.Workload.SeedCount != count {
			t.Fatal("explicit protocol count changed", protocol, err)
		}
		settings.Workload.SeedCount = legacy.Workload.SeedCount
		if !reflect.DeepEqual(settings, legacy) {
			t.Fatal("a non-count setting changed")
		}
	}
}

// A complete composition uses all three newly policy-bound images, exactly one
// native workload generation, one fresh control, and three append-only jobs.
func TestHeldOutFreshSingleCompleteCompositionPreservesOldClaims(t *testing.T) {
	legacy := newHeldOutTwoBCompositionFixture(t)
	if legacy.ConstructorAuthentications != 1 || legacy.Preparation.ConstructorAuthentications != 1 ||
		legacy.Entropy.Calls != 0 || legacy.Preparation.Generated != 0 {
		t.Fatal("ordinary predecessor constructors lost their positive native authentication")
	}
	self := newHeldOutFreshSingleFixture(t)
	op := self.Base.Preparation
	if self.Authentications != 1 || self.Base.ConstructorAuthentications != 0 || op.ConstructorAuthentications != 0 {
		t.Fatal("fresh constructor skipped final authentication or repeated predecessor checks")
	}
	oldStart, err := readHeldOutPinned(self.Declaration.OriginalExecutionStarted, op.Settings.MaximumBytes, true)
	if err != nil {
		t.Fatal(err)
	}
	plan, err := self.prepare(t)
	if err != nil || len(plan.Rounds) != 1 || op.Generated != 1 || self.Rechecks != 3 || self.Authentications != 5 {
		t.Fatal("single-draw preparation failed", err)
	}
	content, err := os.ReadFile(op.PublicPin.Path)
	if err != nil {
		t.Fatal(err)
	}
	var public heldOutPublicPlan
	if decodeHeldOutComplete(content, &public, "synthetic fresh public plan") != nil || public.Protocol != heldOutProtocolFreshSingleB ||
		public.RequiredOrderedSeeds != 1 || public.FreshSingleDeclarationSha256 != op.Config.FreshSingle.Declaration.Sha256 ||
		public.PriorFailedResultSha256 != self.Facts.FailedResult.Sha256 {
		t.Fatal("public new-context lineage missing")
	}
	var private heldOutPrivatePlan
	content, err = readHeldOutPinned(op.PrivatePin, op.Settings.MaximumBytes, true)
	if err != nil || json.Unmarshal(content, &private) != nil {
		t.Fatal("synthetic private plan unavailable", err)
	}
	if bytes.Contains(content, []byte("real production")) {
		t.Fatal("fixture lost synthetic identity")
	}
	result, execution, err := self.execute(t, plan)
	if err != nil || result == nil || !result.Complete || !result.Accepted || result.OrderedSeeds != 1 ||
		len(execution.Invocations) != 3 || len(execution.Controls) != 1 {
		t.Fatal("fresh composed execution failed", err)
	}
	baseline := ""
	for _, outcome := range result.Outcomes {
		if baseline == "" {
			baseline = outcome.BaselineSha256
		}
		if outcome.BaselineSha256 != baseline || len(outcome.CandidateEvaluationIds) != 9 {
			t.Fatal("fresh shared control census changed")
		}
	}
	if _, err := executeHeldOutCampaign(t.Context(), execution.Config, *plan, op.Release, op.Settings, execution.Hooks); err == nil || len(execution.Invocations) != 3 {
		t.Fatal("spent fresh campaign retried")
	}
	after, err := readHeldOutPinned(self.Declaration.OriginalExecutionStarted, op.Settings.MaximumBytes, true)
	if err != nil || !bytes.Equal(oldStart, after) {
		t.Fatal("old execution claim changed")
	}
	if _, err := authenticateHeldOutFreshSingle(op.Config, op.Release, op.Settings, &plan.Margin); err != nil {
		t.Fatal("original second seed changed", err)
	}
}

// A hidden draw is never replaced after an excluded workload or generator failure.
func TestHeldOutFreshSingleGenerationFailureRetainsSpentNamespace(t *testing.T) {
	for _, variant := range []string{"generator", "entropy", "excluded", "pre-entropy-mutation"} {
		self := newHeldOutFreshSingleFixture(t)
		op := self.Base.Preparation
		switch variant {

		}
