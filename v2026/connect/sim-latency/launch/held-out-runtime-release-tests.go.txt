// Reproduces the actual jq sorted map, nullable pristine source and default
// image-tag encodings with a complete synthetic ten-repository source map.
func newHeldOutRuntimeNativeFixture(t *testing.T) (heldOutNativeBuild, heldOutRuntimeSource, heldOutRelease, heldOutRuntimeBuilder) {
	t.Helper()
	runtime, release, bundle, _, _ := newHeldOutRuntimeFixture(t)
	commits := map[string]string{}
	for index, name := range model.CompetitionSourceRepositoryNames() {
		commits[name] = fmt.Sprintf("%040x", index+1)
	}
	release.BaseSha = commits["server"]
	var err error
	release.SourceLockSha256, err = model.CompetitionSourceLockDigest(commits)
	if err != nil {
		t.Fatal(err)
	}
	source := heldOutRuntimeSource{Schema: 1, Kind: "sim-latency-evaluation-source", Temporary: true,
		BaseImageId: release.ImageId, BaseSha: release.BaseSha, Branch: "sim-latency", SourceLockSha256: strings.Repeat("8", 64),
		CandidatePathBase: "mono_root_v1", Repositories: commits}
	build := heldOutNativeBuild{Schema: 1, BaseImageId: release.ImageId, BaseSha: release.BaseSha,
		CandidateSha: strings.Repeat("a", 40), ImageId: "sha256:" + strings.Repeat("9", 64), PatchSha256: strings.Repeat("b", 64),
		PolicySha256: strings.Repeat("c", 64), BuilderSha256: runtime.BuilderSha256, PatchValidatorSha256: runtime.ValidatorSha256,
		OriginalSourceIdentitySha256: heldOutDigest(heldOutRuntimeJson(t, source)), CandidateCommits: map[string]string{}}
	for name, commit := range commits {
		build.CandidateCommits[name] = commit
	}
	build.CandidateCommits["server"] = build.CandidateSha
	content, err := json.MarshalIndent(build.CandidateCommits, "", "  ")
	if err != nil {
		t.Fatal(err)
	}
	build.CandidateCommitsSha256 = heldOutDigest(append(content, '\n'))
	build.ImageKey = heldOutDigest([]byte(strings.Join([]string{build.BaseImageId, build.PatchSha256, build.PolicySha256, build.BuilderSha256}, "\x00")))
	build.Image = "urnetwork/sim-latency-submission:" + build.ImageKey[:32]
	return build, source, release, bundle.Builder
}

// Encoding preserves every explicit false/zero installation field.
func heldOutRuntimeJson(t *testing.T, value any) []byte {
	t.Helper()
	content, err := json.Marshal(value)
	if err != nil {
		t.Fatal(err)
	}
	return content
}

// The selector moves both executable location and all hashes without mutating
// the original release's map or giving B the bridge runner's nested lock.
func TestHeldOutRuntimeSelectsExplicitNewEvaluator(t *testing.T) {
	runtime, _, _, _, _ := newHeldOutRuntimeFixture(t)
	old := heldOutRelease{ScriptDirectory: "/synthetic/old/container", ScriptDigests: map[string]string{"evaluator.sh": strings.Repeat("8", 64)}}
	revised := old
	if err := selectHeldOutAmendedRuntime(&revised, runtime, runtime); err != nil {
		t.Fatal(err)
	}
	program, args, err := heldOutEvaluatorInvocation(revised, "/synthetic/request.json", "/synthetic/result.json")
	if err != nil || program != "/synthetic/control/container/evaluator.sh" || !reflect.DeepEqual(args, []string{"--request", "/synthetic/request.json", "--result", "/synthetic/result.json"}) || old.ScriptDirectory != "/synthetic/old/container" || old.ScriptDigests["evaluator.sh"] != strings.Repeat("8", 64) || revised.Runtime == nil || revised.ScriptDigests["evaluator.sh"] == old.ScriptDigests["evaluator.sh"] {
		t.Fatalf("amended runtime selection/legacy isolation changed: %s %v %v", program, args, err)
	}
	revised.ScriptDirectory = old.ScriptDirectory
	if _, _, err := heldOutEvaluatorInvocation(revised, "/synthetic/request.json", "/synthetic/result.json"); err == nil {
		t.Fatal("C2 release routed through old evaluator")
	}
}

// Reproduces the old selector's refusal of the exact prospective release while
// proving that adding it cannot mutate historical v2 or the measured identity.
func TestHeldOutRuntimeMetadataPreflightSelectsOnlyNamedOverlay(t *testing.T) {
	historical := frozenHeldOutAmendedRuntime()
	metadata := frozenHeldOutMetadataPreflightRuntime()
	oldComponents := heldOutRuntimeComponentDigests()
	original := heldOutRelease{BaseSha: strings.Repeat("1", 40), ImageId: "sha256:" + strings.Repeat("2", 64),
		BinarySha256: strings.Repeat("3", 64), SourceLockSha256: strings.Repeat("4", 64),
		ScriptDirectory: historical.ScriptDirectory, ScriptDigests: oldComponents, Runtime: &historical}
	revised := original
	if err := selectHeldOutAmendedRuntime(&revised, metadata, historical); err != nil {
		t.Fatal("exact named metadata-preflight overlay refused", err)
	}
	if revised.Runtime == nil || !reflect.DeepEqual(*revised.Runtime, metadata) || revised.ScriptDirectory != metadata.ScriptDirectory ||
		!reflect.DeepEqual(revised.ScriptDigests, heldOutMetadataPreflightRuntimeComponentDigests()) ||
		revised.BaseSha != original.BaseSha || revised.ImageId != original.ImageId || revised.BinarySha256 != original.BinarySha256 || revised.SourceLockSha256 != original.SourceLockSha256 ||
		!reflect.DeepEqual(historical, frozenHeldOutAmendedRuntime()) || !reflect.DeepEqual(original.ScriptDigests, heldOutRuntimeComponentDigests()) {
		t.Fatal("metadata selection changed a measured identity or historical installation")
	}
	legacy := original
	if err := selectHeldOutAmendedRuntime(&legacy, historical, historical); err != nil || !reflect.DeepEqual(legacy.Runtime, original.Runtime) || !reflect.DeepEqual(legacy.ScriptDigests, oldComponents) {
		t.Fatal("historical C2 runtime no longer selects its original exact components", err)
	}
}

// Only evaluator and entrypoint change on the host. Maps are private copies;
// no later selection may modify the release already bound to historical v2.
func TestHeldOutRuntimeMetadataPreflightChangesExactlyTwoComponents(t *testing.T) {
	historical := heldOutRuntimeComponentDigests()
	metadata := heldOutMetadataPreflightRuntimeComponentDigests()
	wantChanges := map[string]bool{"evaluator.sh": true, "entrypoint.sh": true}
	if len(historical) != len(metadata) {
		t.Fatal("metadata overlay added or removed a host component")
	}
	for name, digest := range historical {
		if digest != metadata[name] {
			if !wantChanges[name] {
				t.Error("metadata overlay changed an unrelated component", name)
			}
			delete(wantChanges, name)
		}
	}
	if len(wantChanges) != 0 {
		t.Fatal("metadata overlay omitted a required changed component", wantChanges)
	}
	metadata["evaluator.sh"] = strings.Repeat("9", 64)
	if !reflect.DeepEqual(historical, heldOutRuntimeComponentDigests()) || reflect.DeepEqual(metadata, heldOutMetadataPreflightRuntimeComponentDigests()) {
		t.Fatal("component maps alias historical or future runtime selections")
	}
}

// A near-match or a mixture of valid v2/v3 pins is not a third supported runtime.
// Refusals occur in pure selection before reading files or invoking a workload.
func TestHeldOutRuntimeMetadataPreflightRejectsChangedOrMixedPins(t *testing.T) {
	historical, metadata := frozenHeldOutAmendedRuntime(), frozenHeldOutMetadataPreflightRuntime()
	for index, mutate := range []func(*heldOutRuntimeRelease){
		func(r *heldOutRuntimeRelease) { r.ControlRevision = strings.Repeat("9", 40) },
		func(r *heldOutRuntimeRelease) { r.ValidatorSourceRevision = strings.Repeat("9", 40) },
		func(r *heldOutRuntimeRelease) { r.EvidenceMemoryLimitBytes++ },
		func(r *heldOutRuntimeRelease) { r.ScriptDirectory = historical.ScriptDirectory },
		func(r *heldOutRuntimeRelease) { r.Manifest = historical.Manifest },
		func(r *heldOutRuntimeRelease) { r.Manifest.Sha256 = strings.Repeat("9", 64) },
		func(r *heldOutRuntimeRelease) { r.BuilderSha256 = historical.BuilderSha256 },
		func(r *heldOutRuntimeRelease) { r.ValidatorSha256 = strings.Repeat("9", 64) },
		func(r *heldOutRuntimeRelease) { r.Runner = historical.Runner },
		func(r *heldOutRuntimeRelease) { r.Runner.Sha256 = strings.Repeat("9", 64) },
		func(r *heldOutRuntimeRelease) { r.RunnerManifest = historical.RunnerManifest },
		func(r *heldOutRuntimeRelease) { r.RunnerManifest.Sha256 = strings.Repeat("9", 64) },
	} {
		changed := metadata
		mutate(&changed)
		release := heldOutRelease{}
		if err := selectHeldOutAmendedRuntime(&release, changed, historical); err == nil {
			t.Errorf("metadata runtime mutation %d selected", index)
		}
		if _, err := heldOutRuntimeExecutionComponentDigests(changed); err == nil {
			t.Errorf("metadata runtime mutation %d acquired execution components", index)
		}
	}
	for _, runtime := range []heldOutRuntimeRelease{historical, metadata} {
		components, err := heldOutRuntimeExecutionComponentDigests(runtime)
		if err != nil || len(components) != len(heldOutRuntimeComponentDigests()) {
			t.Fatal("an exact named runtime could not recover its closed component set", err)
		}
	}
	changedMap := heldOutMetadataPreflightRuntimeComponentDigests()
	changedMap["evaluator.sh"] = heldOutRuntimeComponentDigests()["evaluator.sh"]
	release := heldOutRelease{Runtime: &metadata, ScriptDirectory: metadata.ScriptDirectory, ScriptDigests: changedMap}
	err := authenticateHeldOutRuntimeFiles(release, defaultHeldOutOperatorSettings())
	if err == nil || err.Error() != "explicit immutable C2 runtime component selection required" {
		t.Fatal("mixed component map reached live installation reads", err)
	}
}

// A complete synthetic lstat chain is independent of the test process's uid.
func newHeldOutRootPathFixture() map[string]heldOutRootPathMetadata {
	return map[string]heldOutRootPathMetadata{
		"/synthetic/control/evaluator.sh": {Mode: 0555, Links: 1},
		"/synthetic/control":              {Mode: os.ModeDir | 0555, Links: 2},
		"/synthetic":                      {Mode: os.ModeDir | 0755, Links: 2},
		"/":                               {Mode: os.ModeDir | 0755, Links: 2},
	}
}

// All ancestors, not just the root-owned leaf, are part of the authority.
func TestHeldOutRuntimeRequiresProtectedRootAncestry(t *testing.T) {
	metadata := newHeldOutRootPathFixture()
	lookup := func(path string) (heldOutRootPathMetadata, error) {
		value, present := metadata[path]
		if !present {
			return value, errors.New("synthetic missing stat")
		}
		return value, nil
	}
	if err := validateHeldOutRootPath("/synthetic/control/evaluator.sh", 0555, lookup); err != nil {
		t.Fatal(err)
	}
	for _, path := range []string{"/synthetic/control/evaluator.sh", "/synthetic/control", "/synthetic", "/"} {
		original := metadata[path]
		changed := original
		changed.Uid = 7
		metadata[path] = changed
		if err := validateHeldOutRootPath("/synthetic/control/evaluator.sh", 0555, lookup); err == nil {
			t.Errorf("non-root path owner accepted: %s", path)
		}
		changed = original
		changed.Mode |= 0020
		metadata[path] = changed
		if err := validateHeldOutRootPath("/synthetic/control/evaluator.sh", 0555, lookup); err == nil {
			t.Errorf("writable ancestry accepted: %s", path)
		}
		metadata[path] = original
	}
}

// Symlinks, extra links, executable-bit changes and absent metadata fail closed.
func TestHeldOutRuntimeRejectsUnsafeRootLeaf(t *testing.T) {
	for index, mutate := range []func(map[string]heldOutRootPathMetadata){func(m map[string]heldOutRootPathMetadata) {
		v := m["/synthetic/control/evaluator.sh"]
		v.Mode |= os.ModeSymlink
		m["/synthetic/control/evaluator.sh"] = v
	}, func(m map[string]heldOutRootPathMetadata) {
		v := m["/synthetic/control/evaluator.sh"]
		v.Links = 2
		m["/synthetic/control/evaluator.sh"] = v
	}, func(m map[string]heldOutRootPathMetadata) {
		v := m["/synthetic/control/evaluator.sh"]
		v.Mode = 0444
		m["/synthetic/control/evaluator.sh"] = v
	}, func(m map[string]heldOutRootPathMetadata) { delete(m, "/synthetic/control") }, func(m map[string]heldOutRootPathMetadata) {
		v := m["/synthetic/control"]
		v.Mode = os.ModeSymlink | 0555
		m["/synthetic/control"] = v
	}} {
		metadata := newHeldOutRootPathFixture()
		mutate(metadata)
		lookup := func(path string) (heldOutRootPathMetadata, error) {
			value, present := metadata[path]
			if !present {
				return value, errors.New("synthetic missing stat")
			}

	pairs[1].Analyzer = heldOutOpWrite(t, pairs[1].Analyzer.Path, []byte("synthetic changed generator implementation\n"))
	if err := authenticateHeldOutGeneratorFiles(pairs, heldOutRelease{Runtime: &heldOutRuntimeRelease{}}, settings); err == nil {

	}
}

// qualification receipts. No measured correctness or direction is claimed.
func newHeldOutBridgePrebuiltFixture(t *testing.T) (heldOutReference, heldOutRequest, heldOutRelease, string) {
	t.Helper()

	root := newCalibrationFileRoot(t)
	patch := heldOutOpWrite(t, filepath.Join(root, "canonical.patch"), []byte("synthetic canonical-reference identity\n"))

		AllowedPaths: []string{"connect/synthetic.go"}, ForbiddenPaths: []string{"connect/sim-latency/**"}}}
	policyBytes, err := json.MarshalIndent(map[string]any{"allowed_paths": request.PatchPolicy.AllowedPaths,

		t.Fatal(err)
	}
	build.PatchSha256, build.PolicySha256 = patch.Sha256, heldOutDigest(append(policyBytes, '\n'))

	original := heldOutOpWrite(t, filepath.Join(root, "evidence", "candidate-original-source.json"), heldOutRuntimeJson(t, source))
	builderPin := heldOutOpWrite(t, filepath.Join(root, "evidence", "builder-identity.json"), heldOutRuntimeJson(t, builder))
	reference := heldOutReference{Role: "no-op", Patch: patch, CandidateSha: build.CandidateSha, ImageId: build.ImageId,
		BinarySha256: strings.Repeat("d", 64), NativeBuild: heldOutOpJson(t, filepath.Join(root, "native-build.json"), build)}
	for _, kind := range []string{"build", "correctness", "direction"} {
		receipt := heldOutReferenceQualification{Schema: 1, Kind: "sim-latency-held-out-reference-" + kind, Role: reference.Role,
			Accepted: true, BaseSha: release.BaseSha, SourceLockSha256: release.SourceLockSha256, BaseImageId: release.ImageId,

// Full native census authentication, not a count-only text check, requires
// exactly the two additional C2 sidecars and retains historical C1 cardinality.
func TestHeldOutBridgeC2CensusRequiresExactNewNativeSidecars(t *testing.T) {
	for _, mutation := range []string{"none", "missing-original", "extra-entry"} {
		input, release, request, settings := newHeldOutBridgeCensusFixture(t)
		release.Runtime = &heldOutRuntimeRelease{}
		path := filepath.Join(input.ArtifactDirectory, "evidence-manifest.json")
		content, err := os.ReadFile(path)
		if err != nil {
			t.Fatal(err)
		}
		var evidence struct {
			Artifacts []calibrationArtifactBinding `json:"artifacts"`
		}
		if err := json.Unmarshal(content, &evidence); err != nil {
			t.Fatal(err)
		}
		names := []string{"evidence/builder-identity.json", "evidence/candidate-original-source.json"}
		if mutation == "missing-original" {
			names[1] = "evidence/synthetic-substitution.json"
		}
		if mutation == "extra-entry" {
			names = append(names, "evidence/synthetic-extra.json")
		}
		for _, name := range names {
			body := []byte("{}\n")
			pin := heldOutOpWrite(t, filepath.Join(input.ArtifactDirectory, name), body)
			evidence.Artifacts = append(evidence.Artifacts, calibrationArtifactBinding{Path: name, Sha256: pin.Sha256, Bytes: int64(len(body))})
		}
		pin := heldOutOpJson(t, path, evidence)
		input.EvidenceManifestSha256 = pin.Sha256
		completionPath := filepath.Join(input.ArtifactDirectory, "evaluation.complete.json")
		content, err = os.ReadFile(completionPath)
		if err != nil {
			t.Fatal(err)
		}
		var complete heldOutCompletion
		if err := json.Unmarshal(content, &complete); err != nil {
			t.Fatal(err)
		}
		complete.Artifacts["evidence_manifest"] = pin.Sha256
		heldOutOpJson(t, completionPath, complete)
		err = authenticateHeldOutBridgeCensus(input, release, request, settings)
		if mutation == "none" && err != nil || mutation != "none" && err == nil {
			t.Fatalf("C2 census %s: %v", mutation, err)
		}
		if mutation == "none" && authenticateHeldOutBridgeCensus(input, heldOutRelease{ImageId: release.ImageId}, request, settings) == nil {
			t.Fatal("historical C1 census silently accepted the new 467-file instrument")
		}
	}
}
