// The early-deployable Mode B is honestly true/true, not a mislabeled disabled
// file, and changes no policy bytes besides its two explicit interface flags.
func TestProductionPolicyPhaseOneSeparatesPreparationEnabledVariant(t *testing.T) {
	original := []byte("# PRIVATE FILE PREPARATION ONLY. Both flags deliberately remain false.\nenabled: false\nautomation_policy:\n  enabled: false\n  source_generation: 0\n")
	prepared, err := prospectiveProductionPreparationVariant(original)
	if err != nil || !bytes.Contains(prepared, []byte("\nenabled: true\n")) || !bytes.Contains(prepared, []byte("\n  enabled: true\n")) {
		t.Fatal("preparation-enabled variant absent", err)
	}
	if bytes.Contains(prepared, []byte("Both flags deliberately remain false")) || !bytes.Contains(prepared, []byte("# MODE B: CONTROL-PREPARATION ENABLED")) {
		t.Fatal("Mode B retained a false disabled-state label")
	}
	restored := bytes.Replace(prepared, []byte("\nenabled: true\n"), []byte("\nenabled: false\n"), 1)
	restored = bytes.Replace(restored, []byte("\n  enabled: true\n"), []byte("\n  enabled: false\n"), 1)
	restored = bytes.Replace(restored, []byte("# MODE B: CONTROL-PREPARATION ENABLED; this file creates no active pointer, open round, worker ACK or ordinary competition worker."), []byte("# PRIVATE FILE PREPARATION ONLY. Both flags deliberately remain false."), 1)
	if !bytes.Equal(original, restored) {
		t.Fatal("Mode B changed more than the two explicit preparation flags")
	}
	if _, err := prospectiveProductionPreparationVariant(prepared); err == nil {
		t.Fatal("already enabled bytes were reinterpreted as a disabled source")
	}
}
